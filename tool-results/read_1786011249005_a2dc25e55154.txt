   528→  const isBusy = agentState === 'thinking' || agentState === 'speaking' || agentState === 'executing';
   529→  const canSend = input.trim().length > 0 && !isBusy && !isListening;
   530→
   531→  // Auto-resize textarea
   532→  useEffect(() => {
   533→    const el = textareaRef.current;
   534→    if (!el) return;
   535→    el.style.height = 'auto';
   536→    el.style.height = Math.min(el.scrollHeight, 120) + 'px';
   537→  }, [input]);

   540→  useEffect(() => {
   541→    return () => {
   542→      if (recognitionRef.current) {
   543→        try { recognitionRef.current.stop(); } catch { /* ignore */ }
   544→      }
   545→    };
   546→  }, []);
   547→
   548→  const handleSend = useCallback(() => {
   549→    if (!canSend) return;
   550→    const text = input.trim();
   551→    setInput('');
   552→    if (textareaRef.current) textareaRef.current.style.height = 'auto';
   553→    sendToJarvis(text);
   554→  }, [canSend, input]);
   555→
   556→  const handleMicToggle = useCallback(() => {
   557→    if (isListening) {
   558→      // Stop recognition
   559→      try { recognitionRef.current?.stop(); } catch { /* ignore */ }
   560→      recognitionRef.current = null;
   561→      setAgentState('idle');
   562→      setTranscript('');
   563→    } else if (!isBusy) {
   564→      // Start STT
   565→      const SpeechRecognitionCtor =
   566→        (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition;
   567→      if (!SpeechRecognitionCtor) return;
   568→
   569→      const recognition = new SpeechRecognitionCtor();
   570→      recognition.continuous = true;
   571→      recognition.interimResults = true;
   572→      recognition.lang = 'en-US';
   573→
   574→      recognition.onresult = (event: any) => {
   575→        let interim = '';
   576→        let final = '';
   577→        for (let i = event.resultIndex; i < event.results.length; i++) {
   578→          const t = event.results[i][0].transcript;
   579→          if (event.results[i].isFinal) {
   580→            final += t;
   581→          } else {
   582→            interim += t;
   583→          }
   584→        }
   585→        if (final.trim()) {
   586→          setTranscript('');
   587→          setAgentState('idle');
   588→          try { recognition.stop(); } catch { /* ignore */ }
   589→          recognitionRef.current = null;
   590→          sendToJarvis(final.trim());
   591→        } else {
   592→          setTranscript(interim);
   593→        }
   594→      };
   595→
   596→      recognition.onerror = () => {
   597→        setAgentState('idle');
   598→        setTranscript('');
   599→        recognitionRef.current = null;
   600→      };
   601→
   602→      recognition.onend = () => {
   603→        // If still in listening state and no final was captured, clean up
   604→        if (useJarvisStore.getState().agentState === 'listening') {
   605→          setAgentState('idle');
   606→          setTranscript('');
   607→          recognitionRef.current = null;
   608→        }
   609→      };
   610→
   611→      recognitionRef.current = recognition;
   612→      try {
   613→        recognition.start();
   614→      } catch {
   615→        // Already started or other error
   616→      }
   617→      setAgentState('listening');
   618→    }
   619→  }, [isListening, isBusy, setAgentState, setTranscript]);
   620→
   621→  const handleImageClick = useCallback(() => {
   622→    fileInputRef.current?.click();
   623→  }, []);
   624→
   625→  const handleImageChange = useCallback(
   626→    (e: React.ChangeEvent<HTMLInputElement>) => {
   627→      const file = e.target.files?.[0];
   628→      if (!file) return;
   629→      const reader = new FileReader();
   630→      reader.onload = () => {
   631→        const base64 = reader.result as string;
   632→        sendToJarvis('What do you see?', base64);
   633→      };
   634→      reader.readAsDataURL(file);
   635→      e.target.value = '';
   636→    },
   637→    [],
   638→  );
   639→
   640→  const handleKeyDown = useCallback(
   641→    (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   642→      if (e.key === 'Enter' && !e.shiftKey) {
   643→        e.preventDefault();
   644→        handleSend();
   645→      }
   646→    },
   647→    [handleSend],
   648→  );
   649→
   650→  return (
   651→    <div className="shrink-0 border-t border-white/[0.06] bg-[#080c16]/80 backdrop-blur-xl px-3 py-3 md:px-4 md:py-3.5">
   652→      <input
   653→        ref={fileInputRef}
   654→        type="file"
   655→        accept="image/*"
   656→        capture="environment"
   657→        className="hidden"
   658→        onChange={handleImageChange}
   659→      />
   660→      <div className="mx-auto flex max-w-3xl items-end gap-2">
   661→        {/* Mic button */}
   662→        <Tooltip>
   663→          <TooltipTrigger asChild>
   664→            <Button
   665→              variant={isListening ? 'destructive' : 'outline'}
   666→              size="icon"
   667→              onClick={handleMicToggle}
   668→              disabled={isBusy && !isListening}
   669→              className={cn(
   670→                'h-10 w-10 shrink-0 rounded-xl border-white/10 bg-white/5 text-white/60 transition-all hover:text-white hover:bg-white/10',
   671→                isListening &&
   672→                  'border-rose-500/40 bg-rose-500/15 text-rose-400 hover:bg-rose-500/25 hover:text-rose-300 animate-pulse',
   673→              )}
   674→            >
   675→              {isListening ? <MicOff className="size-4" /> : <Mic className="size-4" />}
   676→            </Button>
   677→          </TooltipTrigger>
   678→          <TooltipContent side="top" className="text-xs">
   679→            {isListening ? 'Stop listening' : 'Voice input'}
   680→          </TooltipContent>
   681→        </Tooltip>
   682→
   683→        {/* Camera / image button */}
   684→        <Tooltip>
   685→          <TooltipTrigger asChild>
   686→            <Button
   687→              variant="outline"
   688→              size="icon"
   689→              onClick={handleImageClick}
   690→              disabled={isBusy}
   691→              className="h-10 w-10 shrink-0 rounded-xl border-white/10 bg-white/5 text-white/60 transition-all hover:text-white hover:bg-white/10"
   692→            >
   693→              <Camera className="size-4" />
   694→            </Button>
   695→          </TooltipTrigger>
   696→          <TooltipContent side="top" className="text-xs">
   697→            Send image
   698→          </TooltipContent>
   699→        </Tooltip>
   700→
   701→        {/* Text input */}
   702→        <div className="relative flex-1">
   703→          <textarea

   705→            value={input}
   706→            onChange={(e) => setInput(e.target.value)}
   707→            onKeyDown={handleKeyDown}
   708→            placeholder={
   709→              isBusy
   710→                ? agentState === 'thinking'
   711→                  ? 'JARVIS is processing...'
   712→                  : agentState === 'speaking'
   713→                    ? 'JARVIS is speaking...'
   714→                    : 'Executing command...'
   715→                : 'Ask JARVIS anything...'
   716→            }
   717→            disabled={isBusy || isListening}
   718→            rows={1}
   719→            className={cn(
   720→              'w-full resize-none rounded-xl border border-white/[0.08] bg-white/[0.03] px-4 py-2.5 text-sm text-white/90 placeholder:text-white/25 outline-none transition-colors',
   721→              'focus:border-cyan-500/25 focus:bg-white/[0.05]',
   722→              'disabled:cursor-not-allowed disabled:opacity-40',

   725→        </div>

   832→function sendToJarvis(text: string, image?: string) {
   833→  const store = useJarvisStore.getState();
   834→  store.addMessage({
   835→    id: crypto.randomUUID(),
   836→    role: 'user',
   837→    text,
   838→    image: image ? image : undefined,
   839→    timestamp: Date.now(),
   840→  });
   841→  store.setAgentState('thinking');
   842→
   843→  const apiMessages = store.messages
   844→    .filter(m => m.role === 'user' || m.role === 'assistant')
   845→    .slice(-10)
   846→    .map(m => ({ role: m.role, content: m.text }));
   847→
   848→  const body: Record<string, any> = { messages: apiMessages };
   849→  if (image) body.image = image;
   850→
   851→  fetch('/api/jarvis', {
   852→    method: 'POST',
   853→    headers: { 'Content-Type': 'application/json' },
   854→    body: JSON.stringify(body),
   855→  })
   856→    .then(r => r.json())
   857→    .then(data => {
   858→      if (data.error) {
   859→        store.addMessage({ id: crypto.randomUUID(), role: 'system', text: `Error: ${data.error}`, timestamp: Date.now() });
   860→        store.setAgentState('idle');
   861→        return;
   862→      }
   863→      if (data.actionRequest) {
   864→        store.setPendingAction({
   865→          actionId: crypto.randomUUID(),
   866→          tier: data.actionRequest.tier || 'T2',
   867→          tool: data.actionRequest.tool || 'unknown',
   868→          args: data.actionRequest.args || {},
   869→          description: data.actionRequest.description || 'Action requested',
   870→        });
   871→        store.setAgentState('awaiting_confirmation');
   872→        return;
   873→      }
   874→      if (data.text) {
   875→        store.addMessage({ id: crypto.randomUUID(), role: 'assistant', text: data.text, timestamp: Date.now() });
   876→        speakTextWithMouthSync(data.text);
   877→      }
   878→      if (data.toolExecutions) {
   879→        data.toolExecutions.forEach((te: any) =>
   880→          store.setLastToolExec({ tool: te.tool, result: te.result, tier: te.tier }),
   881→        );
   882→      }
   883→      store.setAgentState('idle');
   884→    })
   885→    .catch(() => {
   886→      store.addMessage({ id: crypto.randomUUID(), role: 'system', text: 'Connection error. Check your connection.', timestamp: Date.now() });
   887→      store.setAgentState('idle');
   888→    });
   889→}
   890→
   891→function speakTextWithMouthSync(text: string) {
   892→  const synth = window.speechSynthesis;
   893→  synth.cancel(); // cancel any ongoing speech
   894→
   895→  const store = useJarvisStore.getState();
   896→  store.setAgentState('speaking');
   897→
   898→  // Chunk text into pieces < 150 characters at sentence boundaries
   899→  const chunks: string[] = [];
   900→  const sentences = text.match(/[^.!?\n]+[.!?\n]+/g) || [text];
   901→  let buffer = '';
   902→  for (const s of sentences) {
   903→    if ((buffer + s).length > 150 && buffer.trim()) {
   904→      chunks.push(buffer.trim());
   905→      buffer = s;
   906→    } else {
   907→      buffer += s;
   908→    }
   909→  }
   910→  if (buffer.trim()) chunks.push(buffer.trim());
   911→
   912→  // Mouth animation decay refs
   913→  let mouthDecayTimer: ReturnType<typeof setTimeout> | null = null;
   914→  let keepaliveTimer: ReturnType<typeof setInterval> | null = null;
   915→
   916→  // iOS keepalive: pause/resume every 10s to prevent speech from stopping
   917→  keepaliveTimer = setInterval(() => {
   918→    if (synth.speaking) {
   919→      synth.pause();
   920→      setTimeout(() => synth.resume(), 50);
   921→    }
   922→  }, 10000);
   923→
   924→  const cleanup = () => {
   925→    if (keepaliveTimer) clearInterval(keepaliveTimer);
   926→    if (mouthDecayTimer) clearTimeout(mouthDecayTimer);
   927→    useJarvisStore.getState().setMouthOpen(0);
   928→    useJarvisStore.getState().setAgentState('idle');
   929→  };
   930→
   931→  let chunkIdx = 0;
   932→
   933→  const speakNextChunk = () => {
   934→    if (chunkIdx >= chunks.length) {
   935→      cleanup();
   936→      return;
   937→    }
   938→
   939→    const utt = new SpeechSynthesisUtterance(chunks[chunkIdx]);
   940→
   941→    utt.onboundary = () => {
   942→      // Animate mouth on word boundary
   943→      useJarvisStore.getState().setMouthOpen(0.5);
   944→      if (mouthDecayTimer) clearTimeout(mouthDecayTimer);
   945→      mouthDecayTimer = setTimeout(() => {
   946→        useJarvisStore.getState().setMouthOpen(0);
   947→      }, 200);
   948→    };
   949→
   950→    utt.onend = () => {
   951→      chunkIdx++;
   952→      speakNextChunk();
   953→    };
   954→
   955→    utt.onerror = () => {
   956→      cleanup();
   957→    };
   958→
   959→    synth.speak(utt);
   960→  };
   961→
   962→  speakNextChunk();
   963→}
