    62→    62→    62→function AudioVisualizer({ analyser, active }: { analyser: AnalyserNode | null; active: boolean }) {
    63→    63→    63→  const canvasRef = useRef<HTMLCanvasElement>(null);
    64→    64→    64→  const rafRef = useRef<number>(0);
    65→    65→    65→
    66→    66→    66→  useEffect(() => {
    67→    67→    67→    if (!analyser || !active) return;
    68→    68→    68→    const canvas = canvasRef.current;
    69→    69→    69→    if (!canvas) return;
    70→    70→    70→    const ctx = canvas.getContext('2d');
    71→    71→    71→    if (!ctx) return;
    72→    72→    72→    const bufLen = analyser.frequencyBinCount;
    73→    73→    73→    const data = new Uint8Array(bufLen);
    74→    74→    74→    const barCount = 32;
    75→    75→    75→
    76→    76→    76→    const draw = () => {
    77→    77→    77→      rafRef.current = requestAnimationFrame(draw);
    78→    78→    78→      analyser.getByteFrequencyData(data);
    79→    79→    79→      ctx!.clearRect(0, 0, canvas!.width, canvas!.height);
    80→    80→    80→      const step = Math.floor(bufLen / barCount);
    81→    81→    81→      const barW = canvas!.width / barCount - 1;
    82→    82→    82→      for (let i = 0; i < barCount; i++) {
    83→    83→    83→        const val = data[i * step] / 255;
    84→    84→    84→        const h = val * canvas!.height * 0.9;
    85→    85→    85→        const x = i * (barW + 1);
    86→    86→    86→        const gradient = ctx!.createLinearGradient(0, canvas!.height, 0, canvas!.height - h);
    87→    87→    87→        gradient.addColorStop(0, 'rgba(0, 232, 255, 0.1)');
    88→    88→    88→        gradient.addColorStop(1, `rgba(0, 232, 255, ${0.3 + val * 0.7})`);
    89→    89→    89→        ctx!.fillStyle = gradient;
    90→    90→    90→        ctx!.fillRect(x, canvas!.height - h, barW, h);
    91→    91→    91→      }
    92→    92→    92→    };
    93→    93→    93→    draw();
    94→    94→    94→    return () => cancelAnimationFrame(rafRef.current);
    95→    95→    95→  }, [analyser, active]);
    96→    96→    96→
    97→    97→    97→  return <canvas ref={canvasRef} width={256} height={40} className={cn('w-full h-8 opacity-0 transition-opacity duration-300', active && 'opacity-100')} />;
    98→    98→    98→}

   399→   399→   399→function sendToJarvis(text: string, image?: string) {
   400→   400→   400→  const store = useJarvisStore.getState();
   401→   401→   401→  playSoundEffect('send');
   402→   402→   402→  store.addMessage({ id: crypto.randomUUID(), role: 'user', text, image: image || undefined, timestamp: Date.now() });
   403→   403→   403→  store.setAgentState('thinking');
   404→   404→   404→
   405→   405→   405→  const apiMessages = store.messages
   406→   406→   406→    .filter(m => m.role === 'user' || m.role === 'assistant')
   407→   407→   407→    .slice(-10)
   408→   408→   408→    .map(m => ({ role: m.role, content: m.text }));
   409→   409→   409→
   410→   410→   410→  const body: Record<string, any> = { messages: apiMessages };
   411→   411→   411→  if (image) body.image = image;
   412→   412→   412→
   413→   413→   413→  fetch('/api/jarvis', {
   414→   414→   414→    method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
   415→   415→   415→  })
   416→   416→   416→    .then(r => r.json())
   417→   417→   417→    .then(data => {
   418→   418→   418→      if (data.error) {
   419→   419→   419→        store.addMessage({ id: crypto.randomUUID(), role: 'system', text: `Error: ${data.error}`, timestamp: Date.now() });
   420→   420→   420→        store.setAgentState('idle');
   421→   421→   421→        playSoundEffect('error');
   422→   422→   422→        return;
   423→   423→   423→      }
   424→   424→   424→      if (data.actionRequest) {
   425→   425→   425→        store.setPendingAction({
   426→   426→   426→          actionId: crypto.randomUUID(), tier: data.actionRequest.tier || 'T2',
   427→   427→   427→          tool: data.actionRequest.tool || 'unknown', args: data.actionRequest.args || {},
   428→   428→   428→          description: data.actionRequest.description || 'Action requested',
   429→   429→   429→        });
   430→   430→   430→        store.setAgentState('awaiting_confirmation');
   431→   431→   431→        return;
   432→   432→   432→      }
   433→   433→   433→      if (data.text) {
   434→   434→   434→        store.addMessage({ id: crypto.randomUUID(), role: 'assistant', text: data.text, timestamp: Date.now() });
   435→   435→   435→        playSoundEffect('receive');
   436→   436→   436→        speakTextWithMouthSync(data.text);
   437→   437→   437→      }
   438→   438→   438→      if (data.toolExecutions) {
   439→   439→   439→        data.toolExecutions.forEach((te: any) => store.setLastToolExec({ tool: te.tool, result: te.result, tier: te.tier }));
   440→   440→   440→      }
   441→   441→   441→      // Persist conversation
   442→   442→   442→      const msgs = store.messages.filter(m => m.role === 'user' || m.role === 'assistant');
   443→   443→   443→      saveConversation(msgs.map(m => ({ role: m.role, text: m.text, timestamp: m.timestamp })));
   444→   444→   444→    })
   445→   445→   445→    .catch(() => {
   446→   446→   446→      store.addMessage({ id: crypto.randomUUID(), role: 'system', text: 'Connection error. Check your connection.', timestamp: Date.now() });
   447→   447→   447→      store.setAgentState('idle');
   448→   448→   448→      playSoundEffect('error');
   449→   449→   449→    });
   450→   450→   450→}

   703→   703→   703→function speakTextWithMouthSync(text: string) {
   704→   704→   704→  const synth = window.speechSynthesis;
   705→   705→   705→  synth.cancel();
   706→   706→   706→  const store = useJarvisStore.getState();
   707→   707→   707→  store.setAgentState('speaking');
   708→   708→   708→
   709→   709→   709→  const chunks: string[] = [];
   710→   710→   710→  const sentences = text.match(/[^.!?\n]+[.!?\n]+/g) || [text];
   711→   711→   711→  let buffer = '';
   712→   712→   712→  for (const s of sentences) {
   713→   713→   713→    if ((buffer + s).length > 150 && buffer.trim()) { chunks.push(buffer.trim()); buffer = s; } else { buffer += s; }
   714→   714→   714→  }
   715→   715→   715→  if (buffer.trim()) chunks.push(buffer.trim());
   716→   716→   716→
   717→   717→   717→  let mouthDecay: ReturnType<typeof setTimeout> | null = null;
   718→   718→   718→  let keepalive: ReturnType<typeof setInterval> | null = null;
   719→   719→   719→
   720→   720→   720→  keepalive = setInterval(() => {
   721→   721→   721→    if (synth.speaking) { synth.pause(); setTimeout(() => synth.resume(), 50); }
   722→   722→   722→  }, 10000);
   723→   723→   723→
   724→   724→   724→  const cleanup = () => {
   725→   725→   725→    if (keepalive) clearInterval(keepalive);
   726→   726→   726→    if (mouthDecay) clearTimeout(mouthDecay);
   727→   727→   727→    useJarvisStore.getState().setMouthOpen(0);
   728→   728→   728→    useJarvisStore.getState().setAgentState('idle');
   729→   729→   729→  };
   730→   730→   730→
   731→   731→   731→  let idx = 0;
   732→   732→   732→  const speakNext = () => {
   733→   733→   733→    if (idx >= chunks.length) { cleanup(); return; }
   734→   734→   734→    const utt = new SpeechSynthesisUtterance(chunks[idx]);
   735→   735→   735→    // Select a good English voice
   736→   736→   736→    const voices = synth.getVoices();
   737→   737→   737→    const preferred = voices.find(v => v.lang.startsWith('en') && v.name.includes('Google')) ||
   738→   738→   738→                      voices.find(v => v.lang.startsWith('en') && !v.localService) ||
   739→   739→   739→                      voices.find(v => v.lang.startsWith('en'));
   740→   740→   740→    if (preferred) utt.voice = preferred;
   741→   741→   741→
   742→   742→   742→    utt.onboundary = () => {
   743→   743→   743→      useJarvisStore.getState().setMouthOpen(0.5);
   744→   744→   744→      if (mouthDecay) clearTimeout(mouthDecay);
   745→   745→   745→      mouthDecay = setTimeout(() => { useJarvisStore.getState().setMouthOpen(0); }, 200);
   746→   746→   746→    };
   747→   747→   747→    utt.onend = () => { idx++; speakNext(); };
   748→   748→   748→    utt.onerror = () => { cleanup(); };
   749→   749→   749→    synth.speak(utt);
   750→   750→   750→  };
   751→   751→   751→  speakNext();
   752→   752→   752→}

   761→   761→   761→  const analyserRef = useRef<AnalyserNode | null>(null);
   762→   762→   762→  const bargeInRef = useRef<(() => void) | null>(null);
   763→   763→   763→

   773→   773→   773→    // Register service worker for PWA

   785→   785→   785→      if (tag === 'TEXTAREA' || tag === 'INPUT') return;

   830→   830→   830→            store.setAgentState('listening');
   831→   831→   831→            // Auto-start STT for 15 seconds of continuous listening
   832→   832→   832→            const SR = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition;
   833→   833→   833→            if (!SR) return;
