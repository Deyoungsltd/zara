    61→
    62→function AudioVisualizer({ analyser, active }: { analyser: AnalyserNode | null; active: boolean }) {
    63→  const canvasRef = useRef<HTMLCanvasElement>(null);
    64→  const rafRef = useRef<number>(0);
    65→
    66→  useEffect(() => {
    67→    if (!analyser || !active) return;
    68→    const canvas = canvasRef.current;
    69→    if (!canvas) return;
    70→    const ctx = canvas.getContext('2d');
    71→    if (!ctx) return;
    72→    const bufLen = analyser.frequencyBinCount;
    73→    const data = new Uint8Array(bufLen);
    74→    const barCount = 32;
    75→
    76→    const draw = () => {
    77→      rafRef.current = requestAnimationFrame(draw);
    78→      analyser.getByteFrequencyData(data);
    79→      ctx!.clearRect(0, 0, canvas!.width, canvas!.height);
    80→      const step = Math.floor(bufLen / barCount);
    81→      const barW = canvas!.width / barCount - 1;
    82→      for (let i = 0; i < barCount; i++) {
    83→        const val = data[i * step] / 255;
    84→        const h = val * canvas!.height * 0.9;
    85→        const x = i * (barW + 1);
    86→        const gradient = ctx!.createLinearGradient(0, canvas!.height, 0, canvas!.height - h);
    87→        gradient.addColorStop(0, 'rgba(0, 232, 255, 0.1)');
    88→        gradient.addColorStop(1, `rgba(0, 232, 255, ${0.3 + val * 0.7})`);
    89→        ctx!.fillStyle = gradient;
    90→        ctx!.fillRect(x, canvas!.height - h, barW, h);
    91→      }
    92→    };
    93→    draw();
    94→    return () => cancelAnimationFrame(rafRef.current);
    95→  }, [analyser, active]);
    96→
    97→  return <canvas ref={canvasRef} width={256} height={40} className={cn('w-full h-8 opacity-0 transition-opacity duration-300', active && 'opacity-100')} />;
    98→}

   395→/* ═══════════════════════════════════════════════════════════════
   396→   SEND TO JARVIS (shared)
   397→   ═══════════════════════════════════════════════════════════════ */
   398→
   399→function sendToJarvis(text: string, image?: string) {
   400→  const store = useJarvisStore.getState();
   401→  playSoundEffect('send');
   402→  store.addMessage({ id: crypto.randomUUID(), role: 'user', text, image: image || undefined, timestamp: Date.now() });
   403→  store.setAgentState('thinking');
   404→
   405→  const apiMessages = store.messages
   406→    .filter(m => m.role === 'user' || m.role === 'assistant')
   407→    .slice(-10)
   408→    .map(m => ({ role: m.role, content: m.text }));
   409→
   410→  const body: Record<string, any> = { messages: apiMessages };
   411→  if (image) body.image = image;
   412→
   413→  fetch('/api/jarvis', {
   414→    method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
   415→  })
   416→    .then(r => r.json())
   417→    .then(data => {
   418→      if (data.error) {
   419→        store.addMessage({ id: crypto.randomUUID(), role: 'system', text: `Error: ${data.error}`, timestamp: Date.now() });
   420→        store.setAgentState('idle');
   421→        playSoundEffect('error');
   422→        return;
   423→      }
   424→      if (data.actionRequest) {
   425→        store.setPendingAction({
   426→          actionId: crypto.randomUUID(), tier: data.actionRequest.tier || 'T2',
   427→          tool: data.actionRequest.tool || 'unknown', args: data.actionRequest.args || {},
   428→          description: data.actionRequest.description || 'Action requested',
   429→        });
   430→        store.setAgentState('awaiting_confirmation');
   431→        return;
   432→      }
   433→      if (data.text) {
   434→        store.addMessage({ id: crypto.randomUUID(), role: 'assistant', text: data.text, timestamp: Date.now() });
   435→        playSoundEffect('receive');
   436→        speakTextWithMouthSync(data.text);
   437→      }
   438→      if (data.toolExecutions) {
   439→        data.toolExecutions.forEach((te: any) => store.setLastToolExec({ tool: te.tool, result: te.result, tier: te.tier }));
   440→      }
   441→      // Persist conversation
   442→      const msgs = store.messages.filter(m => m.role === 'user' || m.role === 'assistant');
   443→      saveConversation(msgs.map(m => ({ role: m.role, text: m.text, timestamp: m.timestamp })));
   444→    })
   445→    .catch(() => {
   446→      store.addMessage({ id: crypto.randomUUID(), role: 'system', text: 'Connection error. Check your connection.', timestamp: Date.now() });
   447→      store.setAgentState('idle');
   448→      playSoundEffect('error');
   449→    });
   450→}

   595→              onClick={handleMicToggle} disabled={isBusy && !isListening}

   700→   TTS UTILITY
   701→   ═══════════════════════════════════════════════════════════════ */
   702→
   703→function speakTextWithMouthSync(text: string) {
   704→  const synth = window.speechSynthesis;
   705→  synth.cancel();
   706→  const store = useJarvisStore.getState();
   707→  store.setAgentState('speaking');
   708→
   709→  const chunks: string[] = [];
   710→  const sentences = text.match(/[^.!?\n]+[.!?\n]+/g) || [text];
   711→  let buffer = '';
   712→  for (const s of sentences) {
   713→    if ((buffer + s).length > 150 && buffer.trim()) { chunks.push(buffer.trim()); buffer = s; } else { buffer += s; }
   714→  }
   715→  if (buffer.trim()) chunks.push(buffer.trim());
   716→
   717→  let mouthDecay: ReturnType<typeof setTimeout> | null = null;
   718→  let keepalive: ReturnType<typeof setInterval> | null = null;
   719→
   720→  keepalive = setInterval(() => {
   721→    if (synth.speaking) { synth.pause(); setTimeout(() => synth.resume(), 50); }
   722→  }, 10000);
   723→
   724→  const cleanup = () => {
   725→    if (keepalive) clearInterval(keepalive);
   726→    if (mouthDecay) clearTimeout(mouthDecay);
   727→    useJarvisStore.getState().setMouthOpen(0);
   728→    useJarvisStore.getState().setAgentState('idle');
   729→  };
   730→
   731→  let idx = 0;
   732→  const speakNext = () => {
   733→    if (idx >= chunks.length) { cleanup(); return; }
   734→    const utt = new SpeechSynthesisUtterance(chunks[idx]);
   735→    // Select a good English voice
   736→    const voices = synth.getVoices();
   737→    const preferred = voices.find(v => v.lang.startsWith('en') && v.name.includes('Google')) ||
   738→                      voices.find(v => v.lang.startsWith('en') && !v.localService) ||
   739→                      voices.find(v => v.lang.startsWith('en'));
   740→    if (preferred) utt.voice = preferred;
   741→
   742→    utt.onboundary = () => {
   743→      useJarvisStore.getState().setMouthOpen(0.5);
   744→      if (mouthDecay) clearTimeout(mouthDecay);
   745→      mouthDecay = setTimeout(() => { useJarvisStore.getState().setMouthOpen(0); }, 200);
   746→    };
   747→    utt.onend = () => { idx++; speakNext(); };
   748→    utt.onerror = () => { cleanup(); };
   749→    synth.speak(utt);
   750→  };
   751→  speakNext();
   752→}
