    61→    61→
    62→    62→function AudioVisualizer({ analyser, active }: { analyser: AnalyserNode | null; active: boolean }) {
    63→    63→  const canvasRef = useRef<HTMLCanvasElement>(null);
    64→    64→  const rafRef = useRef<number>(0);
    65→    65→
    66→    66→  useEffect(() => {
    67→    67→    if (!analyser || !active) return;
    68→    68→    const canvas = canvasRef.current;
    69→    69→    if (!canvas) return;
    70→    70→    const ctx = canvas.getContext('2d');
    71→    71→    if (!ctx) return;
    72→    72→    const bufLen = analyser.frequencyBinCount;
    73→    73→    const data = new Uint8Array(bufLen);
    74→    74→    const barCount = 48;
    75→    75→    const centerY = canvas!.height / 2;
    76→    76→
    77→    77→    const draw = () => {
    78→    78→      rafRef.current = requestAnimationFrame(draw);
    79→    79→      analyser.getByteFrequencyData(data);
    80→    80→      ctx!.clearRect(0, 0, canvas!.width, canvas!.height);
    81→    81→
    82→    82→      // Center line at 10% white opacity
    83→    83→      ctx!.strokeStyle = 'rgba(255, 255, 255, 0.1)';
    84→    84→      ctx!.lineWidth = 0.5;
    85→    85→      ctx!.beginPath();
    86→    86→      ctx!.moveTo(0, centerY);
    87→    87→      ctx!.lineTo(canvas!.width, centerY);
    88→    88→      ctx!.stroke();
    89→    89→
    90→    90→      const step = Math.floor(bufLen / barCount);
    91→    91→      const barW = canvas!.width / barCount - 1;
    92→    92→      for (let i = 0; i < barCount; i++) {
    93→    93→        const val = data[i * step] / 255;
    94→    94→        const h = val * centerY * 0.9;
    95→    95→        const x = i * (barW + 1);
    96→    96→        const color = `rgba(0, 232, 255, ${0.3 + val * 0.7})`;
    97→    97→        const gradient = ctx!.createLinearGradient(0, centerY, 0, centerY - h);
    98→    98→        gradient.addColorStop(0, 'rgba(0, 232, 255, 0.1)');
    99→    99→        gradient.addColorStop(1, color);
   100→   100→
   101→   101→        // Glow effect
   102→   102→        ctx!.shadowBlur = 6;
   103→   103→        ctx!.shadowColor = color;
   104→   104→
   105→   105→        ctx!.fillStyle = gradient;
   106→   106→        ctx!.fillRect(x, centerY - h, barW, h);
   107→   107→
   108→   108→        // Reflection below center line at 30% opacity
   109→   109→        ctx!.shadowBlur = 0;
   110→   110→        ctx!.globalAlpha = 0.3;
   111→   111→        const reflGradient = ctx!.createLinearGradient(0, centerY, 0, centerY + h);
   112→   112→        reflGradient.addColorStop(0, 'rgba(0, 232, 255, 0.1)');
   113→   113→        reflGradient.addColorStop(1, 'rgba(0, 232, 255, 0)');
   114→   114→        ctx!.fillStyle = reflGradient;
   115→   115→        ctx!.fillRect(x, centerY, barW, h);
   116→   116→        ctx!.globalAlpha = 1.0;
   117→   117→      }
   118→   118→    };
   119→   119→    draw();
   120→   120→    return () => cancelAnimationFrame(rafRef.current);
   121→   121→  }, [analyser, active]);
   122→   122→
   123→   123→  return <canvas ref={canvasRef} width={384} height={64} className={cn('w-full h-10 opacity-0 transition-opacity duration-300', active && 'opacity-100')} />;
   124→   124→}

   424→   424→
   425→   425→function sendToJarvis(text: string, image?: string) {
   426→   426→  const store = useJarvisStore.getState();
   427→   427→  playSoundEffect('send');
   428→   428→  store.addMessage({ id: crypto.randomUUID(), role: 'user', text, image: image || undefined, timestamp: Date.now() });
   429→   429→  store.setAgentState('thinking');
   430→   430→
   431→   431→  const apiMessages = store.messages
   432→   432→    .filter(m => m.role === 'user' || m.role === 'assistant')
   433→   433→    .slice(-10)
   434→   434→    .map(m => ({ role: m.role, content: m.text }));
   435→   435→
   436→   436→  // Inject memory context as system-like messages at the front
   437→   437→  const memCtx = getMemoryContext();
   438→   438→  if (memCtx) {
   439→   439→    apiMessages.unshift({ role: 'user', content: `[Memory context: ${memCtx}]` });
   440→   440→    apiMessages.unshift({ role: 'assistant', content: 'Understood, I will use this context.' });
   441→   441→  }
   442→   442→
   443→   443→  const body: Record<string, any> = { messages: apiMessages };
   444→   444→  if (image) body.image = image;
   445→   445→
   446→   446→  fetch('/api/jarvis', {
   447→   447→    method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
   448→   448→  })
   449→   449→    .then(r => r.json())
   450→   450→    .then(data => {
   451→   451→      if (data.error) {
   452→   452→        store.addMessage({ id: crypto.randomUUID(), role: 'system', text: `Error: ${data.error}`, timestamp: Date.now() });
   453→   453→        store.setAgentState('idle');
   454→   454→        playSoundEffect('error');
   455→   455→        return;
   456→   456→      }
   457→   457→      if (data.actionRequest) {
   458→   458→        store.setPendingAction({
   459→   459→          actionId: crypto.randomUUID(), tier: data.actionRequest.tier || 'T2',
   460→   460→          tool: data.actionRequest.tool || 'unknown', args: data.actionRequest.args || {},
   461→   461→          description: data.actionRequest.description || 'Action requested',
   462→   462→        });
   463→   463→        store.setAgentState('awaiting_confirmation');
   464→   464→        return;
   465→   465→      }
   466→   466→      if (data.text) {
   467→   467→        store.addMessage({ id: crypto.randomUUID(), role: 'assistant', text: data.text, timestamp: Date.now() });
   468→   468→        playSoundEffect('receive');
   469→   469→        speakTextWithMouthSync(data.text);
   470→   470→      }
   471→   471→      if (data.toolExecutions) {
   472→   472→        data.toolExecutions.forEach((te: any) => store.setLastToolExec({ tool: te.tool, result: te.result, tier: te.tier }));
   473→   473→      }
   474→   474→      // Persist conversation
   475→   475→      const msgs = store.messages.filter(m => m.role === 'user' || m.role === 'assistant');
   476→   476→      saveConversation(msgs.map(m => ({ role: m.role, text: m.text, timestamp: m.timestamp })));
   477→   477→    })
   478→   478→    .catch(() => {
   479→   479→      store.addMessage({ id: crypto.randomUUID(), role: 'system', text: 'Connection error. Check your connection.', timestamp: Date.now() });
   480→   480→      store.setAgentState('idle');
   481→   481→      playSoundEffect('error');
   482→   482→    });
   483→   483→}

   489→   489→function ChatInput() {
   490→   490→  const [input, setInput] = useState('');
   491→   491→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   492→   492→  const recognitionRef = useRef<any>(null);
   493→   493→  const fileInputRef = useRef<HTMLInputElement>(null);
   494→   494→  const dropRef = useRef<HTMLDivElement>(null);
   495→   495→  const [isDragging, setIsDragging] = useState(false);
   496→   496→  const { agentState, setAgentState, setTranscript } = useJarvisStore();
   497→   497→  const isListening = agentState === 'listening';
   498→   498→  const isBusy = agentState === 'thinking' || agentState === 'speaking' || agentState === 'executing';
   499→   499→  const canSend = input.trim().length > 0 && !isBusy && !isListening;
   500→   500→
   501→   501→  // Auto-resize textarea
   502→   502→  useEffect(() => {
   503→   503→    const el = textareaRef.current;
   504→   504→    if (!el) return;
   505→   505→    el.style.height = 'auto';
   506→   506→    el.style.height = Math.min(el.scrollHeight, 120) + 'px';
   507→   507→  }, [input]);
   508→   508→
   509→   509→  // Clipboard paste for images
   510→   510→  useEffect(() => {
   511→   511→    const handlePaste = (e: ClipboardEvent) => {
   512→   512→      const items = e.clipboardData?.items;
   513→   513→      if (!items) return;
   514→   514→      for (const item of Array.from(items)) {
   515→   515→        if (item.type.startsWith('image/')) {
   516→   516→          e.preventDefault();
   517→   517→          const file = item.getAsFile();
   518→   518→          if (file) processImageFile(file);
   519→   519→          return;
   520→   520→        }
   521→   521→      }
   522→   522→    };
   523→   523→    window.addEventListener('paste', handlePaste);
   524→   524→    return () => window.removeEventListener('paste', handlePaste);
   525→   525→  }, [isBusy]);
   526→   526→
   527→   527→  // Drag and drop
   528→   528→  const handleDragOver = useCallback((e: React.DragEvent) => { e.preventDefault(); setIsDragging(true); }, []);
   529→   529→  const handleDragLeave = useCallback(() => setIsDragging(false), []);
   530→   530→  const handleDrop = useCallback((e: React.DragEvent) => {
   531→   531→    e.preventDefault();
   532→   532→    setIsDragging(false);
   533→   533→    const file = e.dataTransfer.files[0];
   534→   534→    if (file && file.type.startsWith('image/')) processImageFile(file);
   535→   535→  }, [isBusy]);
   536→   536→
   537→   537→  function processImageFile(file: File) {
   538→   538→    if (isBusy) return;
   539→   539→    const reader = new FileReader();
   540→   540→    reader.onload = () => { const base64 = reader.result as string; sendToJarvis('What do you see in this image?', base64); };
   541→   541→    reader.readAsDataURL(file);
   542→   542→  }
   543→   543→
   544→   544→  const handleScreenCapture = useCallback(async () => {
   545→   545→    if (isBusy) return;
   546→   546→    try {
   547→   547→      const stream = await navigator.mediaDevices.getDisplayMedia({ video: true });
   548→   548→      const video = document.createElement('video');
   549→   549→      video.srcObject = stream;
   550→   550→      video.muted = true;
   551→   551→      await video.play();
   552→   552→      await new Promise(r => setTimeout(r, 200));
   553→   553→      const canvas = document.createElement('canvas');
   554→   554→      canvas.width = video.videoWidth;
   555→   555→      canvas.height = video.videoHeight;
   556→   556→      canvas.getContext('2d')!.drawImage(video, 0, 0);
   557→   557→      stream.getTracks().forEach(t => t.stop());
   558→   558→      const base64 = canvas.toDataURL('image/png');
   559→   559→      sendToJarvis('What do you see in this screenshot?', base64);
   560→   560→    } catch {
   561→   561→      // User cancelled or browser denied
   562→   562→    }
   563→   563→  }, [isBusy]);
   564→   564→
   565→   565→  // Cleanup recognition
   566→   566→  useEffect(() => {
   567→   567→    return () => { try { recognitionRef.current?.stop(); } catch { /* */ } };
   568→   568→  }, []);
   569→   569→
   570→   570→  const handleSend = useCallback(() => {
   571→   571→    if (!canSend) return;
   572→   572→    const text = input.trim(); setInput('');

   574→   574→    sendToJarvis(text);
   575→   575→  }, [canSend, input]);
   576→   576→
   577→   577→  const handleMicToggle = useCallback(() => {
   578→   578→    if (isListening) {
   579→   579→      try { recognitionRef.current?.stop(); } catch { /* */ }
   580→   580→      recognitionRef.current = null;
   581→   581→      setAgentState('idle'); setTranscript('');
   582→   582→    } else if (!isBusy) {
   583→   583→      const SR = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition;
   584→   584→      if (!SR) return;
   585→   585→      const rec = new SR();
   586→   586→      rec.continuous = true; rec.interimResults = true; rec.lang = 'en-US';
   587→   587→      rec.onresult = (ev: any) => {
   588→   588→        let interim = '', final = '';
   589→   589→        for (let i = ev.resultIndex; i < ev.results.length; i++) {
   590→   590→          const t = ev.results[i][0].transcript;
   591→   591→          if (ev.results[i].isFinal) final += t; else interim += t;
   592→   592→        }
   593→   593→        if (final.trim()) {
   594→   594→          setTranscript(''); setAgentState('idle');
   595→   595→          try { rec.stop(); } catch { /* */ }
   596→   596→          recognitionRef.current = null;
   597→   597→          sendToJarvis(final.trim());
   598→   598→        } else { setTranscript(interim); }
   599→   599→      };
   600→   600→      rec.onerror = () => { setAgentState('idle'); setTranscript(''); recognitionRef.current = null; };
   601→   601→      rec.onend = () => {
   602→   602→        if (useJarvisStore.getState().agentState === 'listening') {
   603→   603→          setAgentState('idle'); setTranscript(''); recognitionRef.current = null;
   604→   604→        }
   605→   605→      };
   606→   606→      recognitionRef.current = rec;
   607→   607→      try { rec.start(); } catch { /* */ }
   608→   608→      setAgentState('listening');
   609→   609→    }
   610→   610→  }, [isListening, isBusy, setAgentState, setTranscript]);
   611→   611→
   612→   612→  const handleKeyDown = useCallback((e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   613→   613→    if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); handleSend(); }
   614→   614→  }, [handleSend]);
   615→   615→
   616→   616→  return (
   617→   617→    <div
   618→   618→      ref={dropRef}
   619→   619→      onDragOver={handleDragOver} onDragLeave={handleDragLeave} onDrop={handleDrop}
   620→   620→      className={cn(
   621→   621→        'shrink-0 border-t bg-[#080c16]/80 backdrop-blur-xl px-3 py-3 md:px-4 md:py-3.5 transition-colors',
   622→   622→        isDragging ? 'border-cyan-500/40 bg-cyan-500/[0.04]' : 'border-white/[0.06]',
   623→   623→      )}
   624→   624→    >
   625→   625→      {/* Drag overlay */}

   627→   627→        {isDragging && (
   628→   628→          <motion.div
   629→   629→            initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   630→   630→            className="absolute inset-0 z-30 flex items-center justify-center rounded-t-xl border-2 border-dashed border-cyan-500/30 bg-cyan-500/[0.03]"
   631→   631→          >
   632→   632→            <div className="flex flex-col items-center gap-2">
   633→   633→              <Upload className="size-6 text-cyan-400" />
   634→   634→              <span className="text-xs text-cyan-300/60" style={{ fontFamily: 'var(--font-orbitron), monospace' }}>DROP IMAGE HERE</span>
   635→   635→            </div>
   636→   636→          </motion.div>
   637→   637→        )}
   638→   638→      </AnimatePresence>
   639→   639→
   640→   640→      <input ref={fileInputRef} type="file" accept="image/*" capture="environment" className="hidden" onChange={(e) => {
   641→   641→        const file = e.target.files?.[0]; if (file) processImageFile(file); e.target.value = '';
   642→   642→      }} />
   643→   643→
   644→   644→      <div className="mx-auto flex max-w-3xl items-end gap-2">
   645→   645→        <Tooltip>
   646→   646→          <TooltipTrigger asChild>
   647→   647→            <Button
   648→   648→              variant={isListening ? 'destructive' : 'outline'} size="icon"
   649→   649→              onClick={handleMicToggle} disabled={isBusy && !isListening}
   650→   650→              data-mic-btn

   652→   652→                'h-10 w-10 shrink-0 rounded-xl border-white/10 bg-white/5 text-white/60 transition-all hover:text-white hover:bg-white/10',
   653→   653→                isListening && 'border-amber-500/40 bg-amber-500/15 text-amber-400 hover:bg-amber-500/25 hover:text-amber-300 animate-pulse',

   750→   750→            {messages.map((msg) => <MessageBubble key={msg.id} message={msg} />)}
   751→   751→            <AnimatePresence>{lastToolExec && <ToolExecutionCard exec={lastToolExec} />}</AnimatePresence>
   752→   752→            <AnimatePresence>{pendingAction && <PendingActionCard action={pendingAction} />}</AnimatePresence>
   753→   753→          </div>
   754→   754→        ) : (
   755→   755→          <WelcomeSuggestions />
   756→   756→        )}
   757→   757→      </ScrollArea>
   758→   758→      <ChatInput />
   759→   759→    </div>
   760→   760→  );
   761→   761→}

   763→   763→/* ═══════════════════════════════════════════════════════════════
   764→   764→   TTS UTILITY
   765→   765→   ═══════════════════════════════════════════════════════════════ */
   766→   766→
   767→   767→function speakTextWithMouthSync(text: string) {
   768→   768→  const synth = window.speechSynthesis;
   769→   769→  synth.cancel();
   770→   770→  const store = useJarvisStore.getState();
   771→   771→  store.setAgentState('speaking');
   772→   772→
   773→   773→  const chunks: string[] = [];
   774→   774→  const sentences = text.match(/[^.!?\n]+[.!?\n]+/g) || [text];
   775→   775→  let buffer = '';
   776→   776→  for (const s of sentences) {
   777→   777→    if ((buffer + s).length > 150 && buffer.trim()) { chunks.push(buffer.trim()); buffer = s; } else { buffer += s; }
   778→   778→  }
   779→   779→  if (buffer.trim()) chunks.push(buffer.trim());
   780→   780→
   781→   781→  let mouthDecay: ReturnType<typeof setTimeout> | null = null;
   782→   782→  let keepalive: ReturnType<typeof setInterval> | null = null;
   783→   783→
   784→   784→  keepalive = setInterval(() => {
   785→   785→    if (synth.speaking) { synth.pause(); setTimeout(() => synth.resume(), 50); }
   786→   786→  }, 10000);
   787→   787→
   788→   788→  const cleanup = () => {
   789→   789→    if (keepalive) clearInterval(keepalive);
   790→   790→    if (mouthDecay) clearTimeout(mouthDecay);
   791→   791→    useJarvisStore.getState().setMouthOpen(0);
   792→   792→    useJarvisStore.getState().setAgentState('idle');
   793→   793→  };
   794→   794→
   795→   795→  let idx = 0;
   796→   796→  const speakNext = () => {
   797→   797→    if (idx >= chunks.length) { cleanup(); return; }
   798→   798→    const utt = new SpeechSynthesisUtterance(chunks[idx]);
   799→   799→    // Select a good English voice
   800→   800→    const voices = synth.getVoices();
   801→   801→    const preferred = voices.find(v => v.lang.startsWith('en') && v.name.includes('Google')) ||
   802→   802→                      voices.find(v => v.lang.startsWith('en') && !v.localService) ||
   803→   803→                      voices.find(v => v.lang.startsWith('en'));
   804→   804→    if (preferred) utt.voice = preferred;
   805→   805→
   806→   806→    utt.onboundary = () => {
   807→   807→      useJarvisStore.getState().setMouthOpen(0.5);
   808→   808→      if (mouthDecay) clearTimeout(mouthDecay);
   809→   809→      mouthDecay = setTimeout(() => { useJarvisStore.getState().setMouthOpen(0); }, 200);
   810→   810→    };
   811→   811→    utt.onend = () => { idx++; speakNext(); };
   812→   812→    utt.onerror = () => { cleanup(); };
   813→   813→    synth.speak(utt);
   814→   814→  };
   815→   815→  speakNext();
   816→   816→}

   822→   822→export default function Home() {
   823→   823→  const [wakeWordActive, setWakeWordActive] = useState(false);
   824→   824→  const wakeWordRef = useRef<WakeWordDetector | null>(null);

   826→   826→  const bargeInRef = useRef<(() => void) | null>(null);
   827→   827→  const backchannelTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   828→   828→
   829→   829→  // Initialize connection + audio + PWA
   830→   830→  useEffect(() => {
   831→   831→    useJarvisStore.getState().setConnected(true);
   832→   832→

   834→   834→    createAudioAnalyser().then(analyser => {
   835→   835→      analyserRef.current = analyser;
   836→   836→    }).catch(() => {});
   837→   837→
   838→   838→    // Register service worker for PWA
   839→   839→    if ('serviceWorker' in navigator) {
   840→   840→      navigator.serviceWorker.register('/sw.js').catch(() => {});

   845→   845→    speechSynthesis.onvoiceschanged = () => speechSynthesis.getVoices();
   846→   846→
   847→   847→    // Keyboard shortcut: Space to toggle mic (when not in textarea)
   848→   848→    const handleKeyDown = (e: KeyboardEvent) => {
   849→   849→      const tag = (e.target as HTMLElement).tagName;
   850→   850→      if (tag === 'TEXTAREA' || tag === 'INPUT') return;
   851→   851→      if (e.code === 'Space') {

   855→   855→      }
   856→   856→    };
   857→   857→    window.addEventListener('keydown', handleKeyDown);
   858→   858→

   878→   878→  });
   879→   879→

   891→   891→        if (state !== 'listening') { active = false; return; }
   892→   892→

   894→   894→

   905→   905→                          voices.find(v => v.lang.startsWith('en') && !v.localService) ||
   906→   906→                          voices.find(v => v.lang.startsWith('en'));

   918→   918→        clearTimeout(backchannelTimerRef.current);
   919→   919→        backchannelTimerRef.current = null;
   920→   920→      }

   926→   926→        active = true;
   927→   927→        scheduleBackchannel();
   928→   928→      } else if (state.agentState !== 'listening' && prev.agentState === 'listening') {
   929→   929→        stopBackchannel();
   930→   930→      }
   931→   931→    });
   932→   932→
   933→   933→    return () => {
   934→   934→      stopBackchannel();
   935→   935→      unsub();
   936→   936→    };

  1001→  1001→    </div>
