    33→function ellipsePoints(
    34→  cx: number, cy: number, rx: number, ry: number, n: number,
    35→  startAngle = 0, endAngle = Math.PI * 2,
    36→): [number, number, number][] {
    37→  const pts: [number, number, number][] = [];
    38→  for (let i = 0; i <= n; i++) {
    39→    const a = startAngle + (endAngle - startAngle) * (i / n);
    40→    pts.push([cx + Math.cos(a) * rx, cy + Math.sin(a) * ry, 0]);
    41→  }
    42→  return pts;
    43→}

    90→function computeHeadOutline(): [number, number, number][] {
    91→  const pts: [number, number, number][] = [];
    92→  for (let i = 0; i <= 64; i++) {
    93→    const a = (i / 64) * Math.PI * 2;
    94→    let r = 1.3;
    95→    const y = Math.sin(a);
    96→    if (y < 0) r += y * 0.25;
    97→    if (Math.abs(Math.cos(a)) < 0.9) r += 0.06;
    98→    if (y > 0.7) r += 0.04;
    99→    pts.push([Math.cos(a) * r, Math.sin(a) * r * 1.15, 0]);
   100→  }
   101→  return pts;
   102→}

   249→/* ================================================================
   250→   CIRCUIT ENERGY PULSE – Traveling energy along circuit paths (shader)
   251→   ================================================================ */
   252→
   253→function CircuitEnergyPulse() {
   254→  const matRef = useRef<THREE.ShaderMaterial>(null);
   255→  const colorObj = useMemo(() => new THREE.Color(), []);
   256→
   257→  const geometry = useMemo(() => {
   258→    const positions: number[] = [];
   259→    const progress: number[] = [];
   260→
   261→    for (const path of CIRCUIT_PATHS) {
   262→      if (path.length < 2) continue;
   263→      // Cumulative distances
   264→      const dists: number[] = [0];
   265→      for (let i = 1; i < path.length; i++) {
   266→        const dx = path[i][0] - path[i - 1][0];
   267→        const dy = path[i][1] - path[i - 1][1];
   268→        const dz = path[i][2] - path[i - 1][2];
   269→        dists.push(dists[i - 1] + Math.sqrt(dx * dx + dy * dy + dz * dz));
   270→      }
   271→      const totalLen = dists[dists.length - 1] || 1;
   272→      // Create line segments
   273→      for (let i = 0; i < path.length - 1; i++) {
   274→        positions.push(path[i][0], path[i][1], path[i][2]);
   275→        progress.push(dists[i] / totalLen);
   276→        positions.push(path[i + 1][0], path[i + 1][1], path[i + 1][2]);
   277→        progress.push(dists[i + 1] / totalLen);
   278→      }
   279→    }
   280→
   281→    const geo = new THREE.BufferGeometry();
   282→    geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
   283→    geo.setAttribute('aProgress', new THREE.Float32BufferAttribute(progress, 1));
   284→    return geo;
   285→  }, []);
   286→
   287→  useFrame(({ clock }) => {
   288→    if (!matRef.current) return;
   289→    matRef.current.uniforms.uTime.value = clock.getElapsedTime();
   290→    const state = useJarvisStore.getState().agentState;
   291→    const c = getStateColors(state).circuitry;
   292→    colorObj.set(c);
   293→    matRef.current.uniforms.uColor.value = colorObj;
   294→  });
   295→
   296→  return (
   297→    <lineSegments geometry={geometry}>
   298→      <shaderMaterial
   299→        ref={matRef}
   300→        uniforms={{
   301→          uTime: { value: 0 },
   302→          uColor: { value: new THREE.Color('#00a8c8') },
   303→        }}
   304→        vertexShader={`
   305→          attribute float aProgress;
   306→          uniform float uTime;
   307→          varying float vPulse;
   308→          void main() {
   309→            float wave = fract(aProgress - uTime * 0.1);
   310→            float pulse = pow(max(0.0, 1.0 - abs(wave - 0.5) * 4.0), 6.0);
   311→            vPulse = pulse;
   312→            gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   313→          }
   314→        `}
   315→        fragmentShader={`
   316→          uniform vec3 uColor;
   317→          varying float vPulse;
   318→          void main() {
   319→            vec3 c = uColor * (1.0 + vPulse * 3.0);
   320→            float alpha = vPulse * 0.7;
   321→            if (alpha < 0.01) discard;
   322→            gl_FragColor = vec4(c, alpha);
   323→          }
   324→        `}
   325→        transparent
   326→        depthWrite={false}
   327→        blending={THREE.AdditiveBlending}
   328→      />
   329→    </lineSegments>
   330→  );
   331→}

   352→/* ================================================================
   353→   HEXAGONAL GRID BACKGROUND – Subtle sci-fi wireframe

   355→
   356→function HexGridBackground() {
   357→  const geo = useMemo(() => {
   358→    const positions: number[] = [];
   359→    const size = 8;
   360→    const hexR = 0.35;
   361→    const sqrt3 = Math.sqrt(3);
   362→    const cols = Math.ceil(size / (hexR * 1.5));
   363→    const rows = Math.ceil(size / (hexR * sqrt3));
   364→
   365→    for (let col = -cols; col <= cols; col++) {
   366→      for (let row = -rows; row <= rows; row++) {
   367→        const cx = col * hexR * 1.5;
   368→        const cy = row * hexR * sqrt3 + (Math.abs(col) % 2 === 1 ? hexR * sqrt3 * 0.5 : 0);
   369→        if (Math.abs(cx) > size || Math.abs(cy) > size) continue;
   370→        for (let k = 0; k < 6; k++) {
   371→          const a1 = (k / 6) * Math.PI * 2 + Math.PI / 6;
   372→          const a2 = ((k + 1) / 6) * Math.PI * 2 + Math.PI / 6;
   373→          positions.push(
   374→            cx + Math.cos(a1) * hexR, cy + Math.sin(a1) * hexR, -2.5,
   375→            cx + Math.cos(a2) * hexR, cy + Math.sin(a2) * hexR, -2.5,
   376→          );
   377→        }
   378→      }
   379→    }
   380→
   381→    const g = new THREE.BufferGeometry();
   382→    g.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
   383→    return g;
   384→  }, []);
   385→
   386→  return (
   387→    <lineSegments geometry={geo}>
   388→      <lineBasicMaterial color="#00d4ff" transparent opacity={0.035} depthWrite={false} />
   389→    </lineSegments>
   390→  );
   391→}

   426→
   427→function HeadWireframe() {
   428→  const mouthOpen = useJarvisStore((s) => s.mouthOpen);
   429→  const agentState = useJarvisStore((s) => s.agentState);
   430→
   431→  /* ── Mouse tracking (module-level shared) ───────────────────── */
   432→  useEffect(() => {
   433→    const handler = (e: MouseEvent) => {
   434→      _mouse.x = (e.clientX / window.innerWidth) * 2 - 1;
   435→      _mouse.y = -(e.clientY / window.innerHeight) * 2 + 1;
   436→      _eye.lastMouseMove = performance.now();
   437→    };
   438→    window.addEventListener('mousemove', handler);
   439→    return () => window.removeEventListener('mousemove', handler);
   440→  }, []);
   441→
   442→  /* ── Refs ───────────────────────────────────────────────────── */
   443→  const leftEyeRef = useRef<THREE.Group>(null);
   444→  const rightEyeRef = useRef<THREE.Group>(null);
   445→  const leftIrisRef = useRef<THREE.Group>(null);
   446→  const rightIrisRef = useRef<THREE.Group>(null);
   447→  const leftBrowRef = useRef<THREE.Group>(null);
   448→  const rightBrowRef = useRef<THREE.Group>(null);
   449→
   450→  /* ── Blink + eye tracking + idle look-around in useFrame ────── */
   451→  useFrame(({ clock }, delta) => {
   452→    const t = clock.getElapsedTime();
   453→
   454→    // Blink logic
   455→    if (_blink.active) {
   456→      _blink.elapsed += delta;
   457→      if (_blink.elapsed < 0.075) {
   458→        _blink.scale = 0.05;
   459→      } else {
   460→        _blink.scale = 1;
   461→        _blink.active = false;
   462→        _blink.elapsed = 0;
   463→        _blink.timer = 0;
   464→        _blink.nextBlink = 3 + Math.random() * 2;
   465→      }
   466→    } else {
   467→      _blink.timer += delta;
   468→      if (_blink.timer >= _blink.nextBlink) {
   469→        _blink.active = true;
   470→        _blink.elapsed = 0;
   471→        _blink.scale = 0.05;
   472→      }
   473→    }
   474→
   475→    const bScale = _blink.scale;
   476→
   477→    // Idle look-around: sinusoidal drift when mouse is still
   478→    const isIdle = performance.now() - _eye.lastMouseMove > 2000;
   479→    let targetX: number, targetY: number;
   480→    if (isIdle) {
   481→      targetX = Math.sin(t * 0.3) * 0.025 + Math.sin(t * 0.17 + 2.0) * 0.012;
   482→      targetY = Math.sin(t * 0.2 + 1.5) * 0.015 + Math.cos(t * 0.13 + 0.7) * 0.008;

   484→      targetX = _mouse.x * 0.04;
   485→      targetY = _mouse.y * 0.04;

   492→    if (rightEyeRef.current) rightEyeRef.current.scale.y = bScale;
   493→    if (leftIrisRef.current) { leftIrisRef.current.position.x = _eye.currentX; leftIrisRef.current.position.y = _eye.currentY; }

   731→   EYE GLOW – Bright pupil dots with blink, tracking, thinking pulse

   733→
   734→function EyeGlow() {
   735→  const matRef = useRef<THREE.ShaderMaterial>(null);
   736→  const geo = useMemo(() => {
   737→    const g = new THREE.BufferGeometry();
   738→    g.setAttribute('position', new THREE.Float32BufferAttribute(
   739→      [[-0.38, 0.25, 0.06], [0.38, 0.25, 0.06]].flat(), 3,
   740→    ));
   741→    return g;
   742→  }, []);
   743→
   744→  const colorObj = useMemo(() => new THREE.Color(), []);
   745→
   746→  useFrame(({ clock }) => {
   747→    if (!matRef.current) return;
   748→    const t = clock.getElapsedTime();
   749→    matRef.current.uniforms.uTime.value = t;
   750→    const state = useJarvisStore.getState().agentState;
   751→    colorObj.set(getStateColors(state).eye);
   752→    matRef.current.uniforms.uColor.value = colorObj;
   753→    matRef.current.uniforms.uBlink.value = _blink.scale;
   754→    matRef.current.uniforms.uMouseX.value = _eye.currentX;
   755→    matRef.current.uniforms.uMouseY.value = _eye.currentY;
   756→    // Thinking pulse
   757→    const thinkingPulse = state === 'thinking' ? 1.0 + 0.4 * Math.sin(t * 3.0) : 1.0;
   758→    matRef.current.uniforms.uThinkingPulse.value = thinkingPulse;
   759→  });
   760→
   761→  return (
   762→    <points geometry={geo}>
   763→      <shaderMaterial
   764→        ref={matRef}
   765→        uniforms={{
   766→          uTime: { value: 0 },
   767→          uColor: { value: new THREE.Color('#00e8ff') },
   768→          uBlink: { value: 1 },
   769→          uMouseX: { value: 0 },
   770→          uMouseY: { value: 0 },
   771→          uThinkingPulse: { value: 1 },
   772→        }}
   773→        vertexShader={`
   774→          uniform float uTime;
   775→          uniform float uMouseX;
   776→          uniform float uMouseY;
   777→          uniform float uThinkingPulse;
   778→          varying float vPulse;
   779→          void main() {
   780→            vec3 p = position;
   781→            p.x += uMouseX;
   782→            p.y += uMouseY;
   783→            vec4 mv = modelViewMatrix * vec4(p, 1.0);
   784→            gl_Position = projectionMatrix * mv;
   785→            gl_PointSize = (8.0 + 2.0 * (uThinkingPulse - 1.0)) * (200.0 / -mv.z);
   786→            vPulse = uThinkingPulse;
   787→          }
   788→        `}
   789→        fragmentShader={`
   790→          uniform vec3 uColor;
   791→          uniform float uBlink;
   792→          varying float vPulse;
   793→          void main() {
   794→            float d = length(gl_PointCoord - vec2(0.5));
   795→            if (d > 0.5) discard;
   796→            float g = 1.0 - smoothstep(0.0, 0.5, d);
   797→            vec3 c = mix(uColor * 0.7, vec3(1.0), pow(g, 2.5));
   798→            c *= (0.85 + 0.15 * vPulse);
   799→            gl_FragColor = vec4(c, g * 0.9 * uBlink);
   800→          }
   801→        `}
   802→        transparent
   803→        depthWrite={false}
   804→        blending={THREE.AdditiveBlending}
   805→      />
   806→    </points>
   807→  );
   808→}

   881→   PARTICLE CLOUD – Holographic particles around head (state color)

   883→
   884→function ParticleCloud() {
   885→  const matRef = useRef<THREE.ShaderMaterial>(null);
   886→  const colorObj = useMemo(() => new THREE.Color(), []);
   887→
   888→  const { geometry, uniforms } = useMemo(() => {
   889→    const count = 1500;
   890→    const pos = new Float32Array(count * 3);
   891→    const sz = new Float32Array(count);
   892→    const rnd = new Float32Array(count);
   893→    for (let i = 0; i < count; i++) {
   894→      const theta = Math.random() * Math.PI * 2;
   895→      const phi = Math.acos(2 * Math.random() - 1);
   896→      const r = 1.35 + Math.random() * 0.5;
   897→      pos[i * 3] = Math.sin(phi) * Math.cos(theta) * r;
   898→      pos[i * 3 + 1] = Math.sin(phi) * Math.sin(theta) * r * 1.15;
   899→      pos[i * 3 + 2] = Math.cos(phi) * r * 0.7;
   900→      sz[i] = Math.random() * 1.5 + 0.3;
   901→      rnd[i] = Math.random();
   902→    }
   903→    const geo = new THREE.BufferGeometry();
   904→    geo.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
   905→    geo.setAttribute('aSize', new THREE.Float32BufferAttribute(sz, 1));
   906→    geo.setAttribute('aRandom', new THREE.Float32BufferAttribute(rnd, 1));

  1136→/* ================================================================
  1137→   HUD RING – Rotating HUD ring with ticks and arcs (state color)
  1138→   ================================================================ */
  1139→
  1140→function HudRing({
  1141→  radius, tiltX, tiltZ, tickCount, majorEvery, speed, opacity, arcs,
  1142→}: {
  1143→  radius: number; tiltX: number; tiltZ: number; tickCount: number; majorEvery: number; speed: number; opacity: number;
  1144→  arcs?: { start: number; end: number; segments: number }[];
  1145→}) {
  1146→  const ref = useRef<THREE.Group>(null);
  1147→  const agentState = useJarvisStore((s) => s.agentState);
  1148→  const ringColor = getStateColors(agentState).hudRing;
  1149→  const speedMult = agentState === 'thinking' ? 2.5 : 1;
  1150→
  1151→  const { tickGeo, arcGeos } = useMemo(() => {
  1152→    const tp: number[] = [];
  1153→    for (let i = 0; i < tickCount; i++) {
  1154→      const a = (i / tickCount) * Math.PI * 2;
  1155→      const len = i % majorEvery === 0 ? 0.16 : 0.07;
  1156→      const inner = radius - len / 2;
  1157→      const outer = radius + len / 2;
  1158→      tp.push(Math.cos(a) * inner, 0, Math.sin(a) * inner, Math.cos(a) * outer, 0, Math.sin(a) * outer);
  1159→    }
  1160→    const tGeo = new THREE.BufferGeometry();
  1161→    tGeo.setAttribute('position', new THREE.Float32BufferAttribute(tp, 3));
  1162→    const aGeos: THREE.BufferGeometry[] = [];
  1163→    if (arcs) arcs.forEach((arc) => {
  1164→      const ap: number[] = [];
  1165→      const step = (arc.end - arc.start) / arc.segments;
  1166→      for (let j = 0; j < arc.segments; j++) {
  1167→        const a1 = arc.start + step * j;
  1168→        const a2 = arc.start + step * (j + 1);
  1169→        ap.push(Math.cos(a1) * radius, 0, Math.sin(a1) * radius, Math.cos(a2) * radius, 0, Math.sin(a2) * radius);
  1170→      }
  1171→      const g = new THREE.BufferGeometry();
  1172→      g.setAttribute('position', new THREE.Float32BufferAttribute(ap, 3));
  1173→      aGeos.push(g);
  1174→    });
  1175→    return { tickGeo: tGeo, arcGeos: aGeos };
  1176→  }, [radius, tickCount, majorEvery, arcs]);
  1177→
  1178→  useFrame((_, delta) => {
  1179→    if (ref.current) ref.current.rotation.y += delta * speed * speedMult;
  1180→  });
  1181→
  1182→  return (
  1183→    <group ref={ref} rotation={[tiltX, 0, tiltZ]}>
  1184→      <lineSegments geometry={tickGeo}>
  1185→        <lineBasicMaterial color={ringColor} transparent opacity={opacity} blending={THREE.AdditiveBlending} depthWrite={false} />
  1186→      </lineSegments>
  1187→      {arcGeos.map((g, i) => (
  1188→        <lineSegments key={i} geometry={g}>
  1189→          <lineBasicMaterial color={ringColor} transparent opacity={opacity * 0.5} blending={THREE.AdditiveBlending} depthWrite={false} />
  1190→        </lineSegments>
  1191→      ))}
  1192→      <mesh rotation={[Math.PI / 2, 0, 0]}>
  1193→        <ringGeometry args={[radius - 0.005, radius + 0.005, 128]} />
  1194→        <meshBasicMaterial color={ringColor} transparent opacity={opacity * 0.2} side={THREE.DoubleSide} blending={THREE.AdditiveBlending} depthWrite={false} />
  1195→      </mesh>
  1196→    </group>
  1197→  );
  1198→}
