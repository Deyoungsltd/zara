    33→    33→function ellipsePoints(
    34→    34→  cx: number, cy: number, rx: number, ry: number, n: number,
    35→    35→  startAngle = 0, endAngle = Math.PI * 2,
    36→    36→): [number, number, number][] {
    37→    37→  const pts: [number, number, number][] = [];
    38→    38→  for (let i = 0; i <= n; i++) {
    39→    39→    const a = startAngle + (endAngle - startAngle) * (i / n);
    40→    40→    pts.push([cx + Math.cos(a) * rx, cy + Math.sin(a) * ry, 0]);
    41→    41→  }
    42→    42→  return pts;
    43→    43→}

   249→   249→/* ================================================================
   250→   250→   CIRCUIT ENERGY PULSE – Traveling energy along circuit paths (shader)
   251→   251→   ================================================================ */
   252→   252→
   253→   253→function CircuitEnergyPulse() {
   254→   254→  const matRef = useRef<THREE.ShaderMaterial>(null);
   255→   255→  const colorObj = useMemo(() => new THREE.Color(), []);
   256→   256→
   257→   257→  const geometry = useMemo(() => {
   258→   258→    const positions: number[] = [];
   259→   259→    const progress: number[] = [];
   260→   260→
   261→   261→    for (const path of CIRCUIT_PATHS) {
   262→   262→      if (path.length < 2) continue;
   263→   263→      // Cumulative distances
   264→   264→      const dists: number[] = [0];
   265→   265→      for (let i = 1; i < path.length; i++) {
   266→   266→        const dx = path[i][0] - path[i - 1][0];
   267→   267→        const dy = path[i][1] - path[i - 1][1];
   268→   268→        const dz = path[i][2] - path[i - 1][2];
   269→   269→        dists.push(dists[i - 1] + Math.sqrt(dx * dx + dy * dy + dz * dz));
   270→   270→      }
   271→   271→      const totalLen = dists[dists.length - 1] || 1;
   272→   272→      // Create line segments
   273→   273→      for (let i = 0; i < path.length - 1; i++) {
   274→   274→        positions.push(path[i][0], path[i][1], path[i][2]);
   275→   275→        progress.push(dists[i] / totalLen);
   276→   276→        positions.push(path[i + 1][0], path[i + 1][1], path[i + 1][2]);
   277→   277→        progress.push(dists[i + 1] / totalLen);
   278→   278→      }
   279→   279→    }
   280→   280→
   281→   281→    const geo = new THREE.BufferGeometry();
   282→   282→    geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
   283→   283→    geo.setAttribute('aProgress', new THREE.Float32BufferAttribute(progress, 1));
   284→   284→    return geo;
   285→   285→  }, []);
   286→   286→
   287→   287→  useFrame(({ clock }) => {
   288→   288→    if (!matRef.current) return;
   289→   289→    matRef.current.uniforms.uTime.value = clock.getElapsedTime();
   290→   290→    const state = useJarvisStore.getState().agentState;
   291→   291→    const c = getStateColors(state).circuitry;
   292→   292→    colorObj.set(c);
   293→   293→    matRef.current.uniforms.uColor.value = colorObj;
   294→   294→  });
   295→   295→
   296→   296→  return (
   297→   297→    <lineSegments geometry={geometry}>
   298→   298→      <shaderMaterial
   299→   299→        ref={matRef}
   300→   300→        uniforms={{
   301→   301→          uTime: { value: 0 },
   302→   302→          uColor: { value: new THREE.Color('#00a8c8') },
   303→   303→        }}
   304→   304→        vertexShader={`
   305→   305→          attribute float aProgress;
   306→   306→          uniform float uTime;
   307→   307→          varying float vPulse;
   308→   308→          void main() {
   309→   309→            float wave = fract(aProgress - uTime * 0.1);
   310→   310→            float pulse = pow(max(0.0, 1.0 - abs(wave - 0.5) * 4.0), 6.0);
   311→   311→            vPulse = pulse;
   312→   312→            gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   313→   313→          }
   314→   314→        `}
   315→   315→        fragmentShader={`
   316→   316→          uniform vec3 uColor;
   317→   317→          varying float vPulse;
   318→   318→          void main() {
   319→   319→            vec3 c = uColor * (1.0 + vPulse * 3.0);
   320→   320→            float alpha = vPulse * 0.7;
   321→   321→            if (alpha < 0.01) discard;
   322→   322→            gl_FragColor = vec4(c, alpha);
   323→   323→          }
   324→   324→        `}
   325→   325→        transparent
   326→   326→        depthWrite={false}
   327→   327→        blending={THREE.AdditiveBlending}
   328→   328→      />
   329→   329→    </lineSegments>
   330→   330→  );
   331→   331→}

   355→   355→
   356→   356→function HexGridBackground() {
   357→   357→  const geo = useMemo(() => {
   358→   358→    const positions: number[] = [];
   359→   359→    const size = 8;
   360→   360→    const hexR = 0.35;
   361→   361→    const sqrt3 = Math.sqrt(3);
   362→   362→    const cols = Math.ceil(size / (hexR * 1.5));
   363→   363→    const rows = Math.ceil(size / (hexR * sqrt3));
   364→   364→
   365→   365→    for (let col = -cols; col <= cols; col++) {
   366→   366→      for (let row = -rows; row <= rows; row++) {
   367→   367→        const cx = col * hexR * 1.5;
   368→   368→        const cy = row * hexR * sqrt3 + (Math.abs(col) % 2 === 1 ? hexR * sqrt3 * 0.5 : 0);
   369→   369→        if (Math.abs(cx) > size || Math.abs(cy) > size) continue;
   370→   370→        for (let k = 0; k < 6; k++) {
   371→   371→          const a1 = (k / 6) * Math.PI * 2 + Math.PI / 6;
   372→   372→          const a2 = ((k + 1) / 6) * Math.PI * 2 + Math.PI / 6;
   373→   373→          positions.push(
   374→   374→            cx + Math.cos(a1) * hexR, cy + Math.sin(a1) * hexR, -2.5,

  1141→  1141→  radius, tiltX, tiltZ, tickCount, majorEvery, speed, opacity, arcs,
  1142→  1142→}: {
  1143→  1143→  radius: number; tiltX: number; tiltZ: number; tickCount: number; majorEvery: number; speed: number; opacity: number;

  1166→  1166→      for (let j = 0; j < arc.segments; j++) {
  1167→  1167→        const a1 = arc.start + step * j;
  1168→  1168→        const a2 = arc.start + step * (j + 1);
  1169→  1169→        ap.push(Math.cos(a1) * radius, 0, Math.sin(a1) * radius, Math.cos(a2) * radius, 0, Math.sin(a2) * radius);
  1170→  1170→      }
  1171→  1171→      const g = new THREE.BufferGeometry();
  1172→  1172→      g.setAttribute('position', new THREE.Float32BufferAttribute(ap, 3));
  1173→  1173→      aGeos.push(g);
  1174→  1174→    });
  1175→  1175→    return { tickGeo: tGeo, arcGeos: aGeos };
  1176→  1176→  }, [radius, tickCount, majorEvery, arcs]);
  1177→  1177→
  1178→  1178→  useFrame((_, delta) => {
  1179→  1179→    if (ref.current) ref.current.rotation.y += delta * speed * speedMult;
  1180→  1180→  });

  1182→  1182→  return (

  1185→  1185→        <lineBasicMaterial color={ringColor} transparent opacity={opacity} blending={THREE.AdditiveBlending} depthWrite={false} />
  1186→  1186→      </lineSegments>
  1187→  1187→      {arcGeos.map((g, i) => (
  1188→  1188→        <lineSegments key={i} geometry={g}>
  1189→  1189→          <lineBasicMaterial color={ringColor} transparent opacity={opacity * 0.5} blending={THREE.AdditiveBlending} depthWrite={false} />
  1190→  1190→        </lineSegments>
  1191→  1191→      ))}
  1192→  1192→      <mesh rotation={[Math.PI / 2, 0, 0]}>
  1193→  1193→        <ringGeometry args={[radius - 0.005, radius + 0.005, 128]} />
  1194→  1194→        <meshBasicMaterial color={ringColor} transparent opacity={opacity * 0.2} side={THREE.DoubleSide} blending={THREE.AdditiveBlending} depthWrite={false} />
  1195→  1195→      </mesh>
  1196→  1196→    </group>
  1197→  1197→  );
  1198→  1198→}

  1204→  1204→function HudDataReadouts() {
  1205→  1205→  const groupRef = useRef<THREE.Group>(null);
  1206→  1206→  const agentState = useJarvisStore((s) => s.agentState);
  1207→  1207→  const readoutColor = getStateColors(agentState).hudRing;

  1211→  1211→    const count = 10;

  1213→  1213→      const angle = (i / count) * Math.PI * 2 + 0.3;
  1214→  1214→      const r = 2.05 + (i % 3) * 0.08;
  1215→  1215→      const yOff = Math.sin(i * 2.1) * 0.2;

  1218→  1218→        scale: [0.06 + (i % 4) * 0.02, 0.012 + (i % 3) * 0.005],
  1219→  1219→        opacity: 0.15 + (i % 3) * 0.05,
  1220→  1220→      });
  1221→  1221→    }
  1222→  1222→    return items;
  1223→  1223→  }, []);
  1224→  1224→
  1225→  1225→  useFrame(({ clock }) => {
  1226→  1226→    if (groupRef.current) groupRef.current.rotation.y = clock.getElapsedTime() * 0.1;

  1245→  1245→function EnergyWaves() {
  1246→  1246→  const meshRefs = useRef<(THREE.Mesh | null)[]>([]);
  1247→  1247→  const colorObj = useMemo(() => new THREE.Color(), []);
  1248→  1248→  const count = 4;

  1254→  1254→    for (let i = 0; i < count; i++) {
  1255→  1255→      const mesh = meshRefs.current[i];
  1256→  1256→      if (!mesh) continue;
  1257→  1257→      const phase = ((t * 0.12 + i / count) % 1);
  1258→  1258→      const s = 0.8 + phase * 2.8;
  1259→  1259→      mesh.scale.set(s, s, 1);
  1260→  1260→      (mesh.material as THREE.MeshBasicMaterial).color.copy(colorObj);
  1261→  1261→      (mesh.material as THREE.MeshBasicMaterial).opacity = Math.pow(1 - phase, 2.5) * 0.055;
  1262→  1262→    }
  1263→  1263→  });

  1265→  1265→  return (

  1269→  1269→          <ringGeometry args={[0.98, 1.01, 64]} />
  1270→  1270→          <meshBasicMaterial color="#00d4ff" transparent opacity={0} side={THREE.DoubleSide} blending={THREE.AdditiveBlending} depthWrite={false} />

  1314→  1314→function JarvisScene() {

  1319→  1319→  useFrame((state, delta) => {

  1322→  1322→    const t = state.clock.getElapsedTime();
  1323→  1323→
  1324→  1324→    // Base Y rotation speed (faster when thinking)
  1325→  1325→    let ySpeed = agentState === 'thinking' ? 0.25 : agentState === 'speaking' ? 0.08 : 0.05;
  1326→  1326→    baseAngleRef.current += delta * ySpeed;
  1327→  1327→
  1328→  1328→    // Y oscillation (side-to-side sway when speaking)
  1329→  1329→    let yOscillation = 0;
  1330→  1330→    if (agentState === 'speaking') {
  1331→  1331→      yOscillation = 0.06 * Math.sin(t * 1.8);
  1332→  1332→    }
  1333→  1333→
  1334→  1334→    // X rotation (subtle idle oscillation + forward lean when listening)
  1335→  1335→    let xRot = 0.01 * Math.sin(t * 0.5);
  1336→  1336→    if (agentState === 'listening') {
  1337→  1337→      xRot += 0.03;
  1338→  1338→    }
  1339→  1339→
  1340→  1340→    // Z rotation (head tilt when thinking)
  1341→  1341→    let zRot = 0;
  1342→  1342→    if (agentState === 'thinking') {
  1343→  1343→      zRot = 0.03 * Math.sin(t * 1.2);
  1344→  1344→    }
  1345→  1345→
  1346→  1346→    headRef.current.rotation.y = baseAngleRef.current + yOscillation;
  1347→  1347→    headRef.current.rotation.x = xRot;
  1348→  1348→    headRef.current.rotation.z = zRot;
  1349→  1349→
  1350→  1350→    // Audio reactivity – subtle scale pulse
  1351→  1351→    const audioScale = 1 + audioLevel * 0.025;
  1352→  1352→    headRef.current.scale.setScalar(audioScale);

  1362→  1362→          <HeadWireframe />
  1363→  1363→          <EyeGlow />
  1364→  1364→          <EyeHaloGlow />
  1365→  1365→          <CircuitEnergyPulse />
  1366→  1366→          <ParticleCloud />
  1367→  1367→          <DataStreamParticles />
  1368→  1368→          <HudRing
  1369→  1369→            radius={1.85} tiltX={0.12} tiltZ={0.04} tickCount={72} majorEvery={6} speed={0.1} opacity={0.3}
  1370→  1370→            arcs={[
