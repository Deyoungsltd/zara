    66→
    67→function AudioVisualizer({ analyser, active }: { analyser: AnalyserNode | null; active: boolean }) {
    68→  const canvasRef = useRef<HTMLCanvasElement>(null);
    69→  const rafRef = useRef<number>(0);
    70→
    71→  useEffect(() => {
    72→    if (!analyser || !active) return;
    73→    const canvas = canvasRef.current;
    74→    if (!canvas) return;
    75→    const ctx = canvas.getContext('2d');
    76→    if (!ctx) return;
    77→    const bufLen = analyser.frequencyBinCount;
    78→    const data = new Uint8Array(bufLen);
    79→    const barCount = 48;
    80→    const centerY = canvas!.height / 2;
    81→
    82→    const draw = () => {
    83→      rafRef.current = requestAnimationFrame(draw);
    84→      analyser.getByteFrequencyData(data);
    85→      ctx!.clearRect(0, 0, canvas!.width, canvas!.height);
    86→
    87→      // Center line at 10% white opacity
    88→      ctx!.strokeStyle = 'rgba(255, 255, 255, 0.1)';
    89→      ctx!.lineWidth = 0.5;
    90→      ctx!.beginPath();
    91→      ctx!.moveTo(0, centerY);
    92→      ctx!.lineTo(canvas!.width, centerY);
    93→      ctx!.stroke();
    94→
    95→      const step = Math.floor(bufLen / barCount);
    96→      const barW = canvas!.width / barCount - 1;
    97→      for (let i = 0; i < barCount; i++) {
    98→        const val = data[i * step] / 255;
    99→        const h = val * centerY * 0.9;
   100→        const x = i * (barW + 1);
   101→        const color = `rgba(0, 232, 255, ${0.3 + val * 0.7})`;
   102→        const gradient = ctx!.createLinearGradient(0, centerY, 0, centerY - h);
   103→        gradient.addColorStop(0, 'rgba(0, 232, 255, 0.1)');
   104→        gradient.addColorStop(1, color);
   105→
   106→        // Glow effect
   107→        ctx!.shadowBlur = 6;
   108→        ctx!.shadowColor = color;
   109→
   110→        ctx!.fillStyle = gradient;
   111→        ctx!.fillRect(x, centerY - h, barW, h);
   112→
   113→        // Reflection below center line at 30% opacity
   114→        ctx!.shadowBlur = 0;
   115→        ctx!.globalAlpha = 0.3;
   116→        const reflGradient = ctx!.createLinearGradient(0, centerY, 0, centerY + h);
   117→        reflGradient.addColorStop(0, 'rgba(0, 232, 255, 0.1)');
   118→        reflGradient.addColorStop(1, 'rgba(0, 232, 255, 0)');
   119→        ctx!.fillStyle = reflGradient;
   120→        ctx!.fillRect(x, centerY, barW, h);
   121→        ctx!.globalAlpha = 1.0;
   122→      }
   123→    };
   124→    draw();
   125→    return () => cancelAnimationFrame(rafRef.current);
   126→  }, [analyser, active]);
   127→
   128→  return <canvas ref={canvasRef} width={384} height={64} className={cn('w-full h-10 opacity-0 transition-opacity duration-300', active && 'opacity-100')} />;
   129→}

   388→/* ═══════════════════════════════════════════════════════════════
   389→   PENDING ACTION CARD
   390→   ═══════════════════════════════════════════════════════════════ */
   391→
   392→function PendingActionCard({ action }: { action: PendingAction }) {
   393→  const { setPendingAction } = useJarvisStore();
   394→  const [verifying, setVerifying] = useState(false);
   395→  const [verifyError, setVerifyError] = useState('');
   396→  const handleConfirm = useCallback(async () => {
   397→    // Security verification check
   398→    if (requireVerification()) {
   399→      setVerifying(true);
   400→      setVerifyError('');
   401→      const level = getSecurityLevel();
   402→      let verified = false;
   403→
   404→      if (level === 'biometric' || level === 'both') {
   405→        verified = await verifyWithBiometric();
   406→      }
   407→      if (!verified && (level === 'voice' || level === 'both')) {
   408→        verified = await verifyWithVoice();
   409→      }
   410→
   411→      setVerifying(false);
   412→      if (!verified) {
   413→        setVerifyError('Verification failed. Please try again.');
   414→        playSoundEffect('error');
   415→        return;
   416→      }
   417→    }
   418→    setPendingAction(null); sendToJarvis('approved');
   419→  }, [setPendingAction]);
   420→  const handleDeny = useCallback(() => { setPendingAction(null); sendToJarvis('cancelled'); }, [setPendingAction]);
   421→
   422→  return (
   423→    <motion.div initial={{ opacity: 0, y: 10, scale: 0.98 }} animate={{ opacity: 1, y: 0, scale: 1 }} className="mx-4">
   424→      <div className="rounded-xl border border-orange-500/20 bg-orange-500/[0.06] p-4">
   425→        <div className="flex items-center gap-2 mb-3">
   426→          <ShieldAlert className="size-4 text-orange-400" />
   427→          <span className="text-xs font-semibold text-orange-400">Action Requires Confirmation</span>
   428→          <Badge variant="outline" className="ml-auto border-white/10 bg-white/5 text-[10px] text-white/50">Tier {action.tier}</Badge>
   429→        </div>
   430→        <div className="space-y-2 mb-4">
   431→          <div className="flex items-center gap-2">
   432→            <Terminal className="size-3 text-white/30" />
   433→            <span className="text-xs font-mono text-orange-300/90" style={{ fontFamily: 'var(--font-orbitron), monospace' }}>{action.tool}</span>
   434→          </div>
   435→          <p className="text-sm text-white/70 leading-relaxed">{action.description}</p>
   436→          {Object.keys(action.args).length > 0 && (
   437→            <pre className="rounded-lg bg-black/30 px-3 py-2 text-xs font-mono text-white/50 overflow-x-auto">{JSON.stringify(action.args, null, 2)}</pre>
   438→          )}
   439→        </div>
   440→        {verifyError && (
   441→          <div className="flex items-center gap-2 mb-3 rounded-lg bg-red-500/10 border border-red-500/20 px-3 py-2">
   442→            <X className="size-3 text-red-400 shrink-0" />
   443→            <span className="text-xs text-red-300/80">{verifyError}</span>
   444→          </div>
   445→        )}
   446→        <Separator className="bg-orange-500/10 mb-3" />
   447→        <div className="flex items-center gap-2 justify-end">
   448→          <Button variant="ghost" size="sm" onClick={handleDeny} className="h-8 gap-1.5 text-white/50 hover:text-rose-300 hover:bg-rose-500/10">

   502→/* ═══════════════════════════════════════════════════════════════
   503→   SEND TO JARVIS (shared)
   504→   ═══════════════════════════════════════════════════════════════ */
   505→
   506→function sendToJarvis(text: string, image?: string) {
   507→  const store = useJarvisStore.getState();
   508→  playSoundEffect('send');
   509→  store.addMessage({ id: crypto.randomUUID(), role: 'user', text, image: image || undefined, timestamp: Date.now() });
   510→  store.setAgentState('thinking');
   511→
   512→  const apiMessages = store.messages
   513→    .filter(m => m.role === 'user' || m.role === 'assistant')
   514→    .slice(-10)
   515→    .map(m => ({ role: m.role, content: m.text }));
   516→
   517→  // Inject memory context as system-like messages at the front
   518→  const memCtx = getMemoryContext();
   519→  if (memCtx) {
   520→    apiMessages.unshift({ role: 'user', content: `[Memory context: ${memCtx}]` });
   521→    apiMessages.unshift({ role: 'assistant', content: 'Understood, I will use this context.' });
   522→  }
   523→
   524→  const body: Record<string, any> = { messages: apiMessages };
   525→  if (image) body.image = image;
   526→
   527→  fetch('/api/jarvis', {
   528→    method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
   529→  })
   530→    .then(r => r.json())
   531→    .then(data => {
   532→      if (data.error) {
   533→        store.addMessage({ id: crypto.randomUUID(), role: 'system', text: `Error: ${data.error}`, timestamp: Date.now() });
   534→        store.setAgentState('idle');
   535→        playSoundEffect('error');
   536→        return;
   537→      }
   538→      if (data.actionRequest) {
   539→        store.setPendingAction({
   540→          actionId: crypto.randomUUID(), tier: data.actionRequest.tier || 'T2',
   541→          tool: data.actionRequest.tool || 'unknown', args: data.actionRequest.args || {},
   542→          description: data.actionRequest.description || 'Action requested',
   543→        });
   544→        store.setAgentState('awaiting_confirmation');
   545→        return;
   546→      }
   547→      if (data.text) {
   548→        store.addMessage({ id: crypto.randomUUID(), role: 'assistant', text: data.text, timestamp: Date.now() });
   549→        playSoundEffect('receive');
   550→        speakTextWithMouthSync(data.text);
   551→      }
   552→      if (data.toolExecutions) {
   553→        data.toolExecutions.forEach((te: any) => store.setLastToolExec({ tool: te.tool, result: te.result, tier: te.tier }));
   554→      }
   555→      // Persist conversation
   556→      const msgs = store.messages.filter(m => m.role === 'user' || m.role === 'assistant');
   557→      saveConversation(msgs.map(m => ({ role: m.role, text: m.text, timestamp: m.timestamp })));
   558→    })
   559→    .catch(() => {
   560→      store.addMessage({ id: crypto.randomUUID(), role: 'system', text: 'Connection error. Check your connection.', timestamp: Date.now() });
   561→      store.setAgentState('idle');
   562→      playSoundEffect('error');
   563→    });
   564→}

   570→function ChatInput() {
   571→  const [input, setInput] = useState('');
   572→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   573→  const recognitionRef = useRef<any>(null);
   574→  const fileInputRef = useRef<HTMLInputElement>(null);
   575→  const dropRef = useRef<HTMLDivElement>(null);
   576→  const [isDragging, setIsDragging] = useState(false);
   577→  const { agentState, setAgentState, setTranscript } = useJarvisStore();
   578→  const isListening = agentState === 'listening';
   579→  const isBusy = agentState === 'thinking' || agentState === 'speaking' || agentState === 'executing';
   580→  const canSend = input.trim().length > 0 && !isBusy && !isListening;
   581→
   582→  // Auto-resize textarea
   583→  useEffect(() => {
   584→    const el = textareaRef.current;
   585→    if (!el) return;
   586→    el.style.height = 'auto';
   587→    el.style.height = Math.min(el.scrollHeight, 120) + 'px';
   588→  }, [input]);
   589→
   590→  // Clipboard paste for images
   591→  useEffect(() => {
   592→    const handlePaste = (e: ClipboardEvent) => {
   593→      const items = e.clipboardData?.items;
   594→      if (!items) return;
   595→      for (const item of Array.from(items)) {
   596→        if (item.type.startsWith('image/')) {
   597→          e.preventDefault();
   598→          const file = item.getAsFile();
   599→          if (file) processImageFile(file);
   600→          return;
   601→        }
   602→      }
   603→    };
   604→    window.addEventListener('paste', handlePaste);
   605→    return () => window.removeEventListener('paste', handlePaste);
   606→  }, [isBusy]);
   607→
   608→  // Drag and drop
   609→  const handleDragOver = useCallback((e: React.DragEvent) => { e.preventDefault(); setIsDragging(true); }, []);
   610→  const handleDragLeave = useCallback(() => setIsDragging(false), []);
   611→  const handleDrop = useCallback((e: React.DragEvent) => {
   612→    e.preventDefault();
   613→    setIsDragging(false);
   614→    const file = e.dataTransfer.files[0];
   615→    if (file && file.type.startsWith('image/')) processImageFile(file);
   616→  }, [isBusy]);

   618→  function processImageFile(file: File) {
   619→    if (isBusy) return;
   620→    const reader = new FileReader();
   621→    reader.onload = () => { const base64 = reader.result as string; sendToJarvis('What do you see in this image?', base64); };
   622→    reader.readAsDataURL(file);
   623→  }
   624→
   625→  const handleScreenCapture = useCallback(async () => {
   626→    if (isBusy) return;
   627→    try {
   628→      const stream = await navigator.mediaDevices.getDisplayMedia({ video: true });
   629→      const video = document.createElement('video');
   630→      video.srcObject = stream;
   631→      video.muted = true;
   632→      await video.play();
   633→      await new Promise(r => setTimeout(r, 200));
   634→      const canvas = document.createElement('canvas');
   635→      canvas.width = video.videoWidth;
   636→      canvas.height = video.videoHeight;
   637→      canvas.getContext('2d')!.drawImage(video, 0, 0);
   638→      stream.getTracks().forEach(t => t.stop());
   639→      const base64 = canvas.toDataURL('image/png');
   640→      sendToJarvis('What do you see in this screenshot?', base64);
   641→    } catch {
   642→      // User cancelled or browser denied
   643→    }
   644→  }, [isBusy]);
   645→
   646→  // Cleanup recognition
   647→  useEffect(() => {
   648→    return () => { try { recognitionRef.current?.stop(); } catch { /* */ } };
   649→  }, []);
   650→

   652→    if (!canSend) return;
   653→    const text = input.trim(); setInput('');

   660→      try { recognitionRef.current?.stop(); } catch { /* */ }
   661→      recognitionRef.current = null;
   662→      setAgentState('idle'); setTranscript('');
   663→    } else if (!isBusy) {

   665→      if (!SR) return;
   666→      const rec = new SR();
   667→      rec.continuous = true; rec.interimResults = true; rec.lang = 'en-US';
   668→      rec.onresult = (ev: any) => {
   669→        let interim = '', final = '';
   670→        for (let i = ev.resultIndex; i < ev.results.length; i++) {
   671→          const t = ev.results[i][0].transcript;
   672→          if (ev.results[i].isFinal) final += t; else interim += t;
   673→        }
   674→        if (final.trim()) {
   675→          setTranscript(''); setAgentState('idle');
   676→          try { rec.stop(); } catch { /* */ }
   677→          recognitionRef.current = null;
   678→          sendToJarvis(final.trim());
   679→        } else { setTranscript(interim); }
   680→      };
   681→      rec.onerror = () => { setAgentState('idle'); setTranscript(''); recognitionRef.current = null; };
   682→      rec.onend = () => {
   683→        if (useJarvisStore.getState().agentState === 'listening') {
   684→          setAgentState('idle'); setTranscript(''); recognitionRef.current = null;
   685→        }
   686→      };
   687→      recognitionRef.current = rec;
   688→      try { rec.start(); } catch { /* */ }
   689→      setAgentState('listening');
   690→    }

   694→    if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); handleSend(); }
   695→  }, [handleSend]);

   699→      ref={dropRef}
   700→      onDragOver={handleDragOver} onDragLeave={handleDragLeave} onDrop={handleDrop}

   706→      {/* Drag overlay */}
   707→      <AnimatePresence>
   708→        {isDragging && (
   709→          <motion.div
   710→            initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   711→            className="absolute inset-0 z-30 flex items-center justify-center rounded-t-xl border-2 border-dashed border-cyan-500/30 bg-cyan-500/[0.03]"
   712→          >
   713→            <div className="flex flex-col items-center gap-2">
   714→              <Upload className="size-6 text-cyan-400" />
   715→              <span className="text-xs text-cyan-300/60" style={{ fontFamily: 'var(--font-orbitron), monospace' }}>DROP IMAGE HERE</span>
   716→            </div>
   717→          </motion.div>
   718→        )}
   719→      </AnimatePresence>
   720→
   721→      <input ref={fileInputRef} type="file" accept="image/*" capture="environment" className="hidden" onChange={(e) => {
   722→        const file = e.target.files?.[0]; if (file) processImageFile(file); e.target.value = '';

   725→      <div className="mx-auto flex max-w-3xl items-end gap-2">

   737→              {isListening ? <MicOff className="size-4" /> : <Mic className="size-4" />}

   739→          </TooltipTrigger>
   740→          <TooltipContent side="top" className="text-xs">{isListening ? 'Stop' : 'Voice (or press Space)'}</TooltipContent>

   742→

   845→   TTS UTILITY
   846→   ═══════════════════════════════════════════════════════════════ */
   847→
   848→function speakTextWithMouthSync(text: string) {
   849→  const synth = window.speechSynthesis;
   850→  synth.cancel();
   851→  const store = useJarvisStore.getState();
   852→  store.setAgentState('speaking');
   853→
   854→  const chunks: string[] = [];
   855→  const sentences = text.match(/[^.!?\n]+[.!?\n]+/g) || [text];
   856→  let buffer = '';
   857→  for (const s of sentences) {
   858→    if ((buffer + s).length > 150 && buffer.trim()) { chunks.push(buffer.trim()); buffer = s; } else { buffer += s; }
   859→  }
   860→  if (buffer.trim()) chunks.push(buffer.trim());
   861→
   862→  let mouthDecay: ReturnType<typeof setTimeout> | null = null;
   863→  let keepalive: ReturnType<typeof setInterval> | null = null;
   864→
   865→  keepalive = setInterval(() => {
   866→    if (synth.speaking) { synth.pause(); setTimeout(() => synth.resume(), 50); }
   867→  }, 10000);
   868→
   869→  const cleanup = () => {
   870→    if (keepalive) clearInterval(keepalive);
   871→    if (mouthDecay) clearTimeout(mouthDecay);
   872→    useJarvisStore.getState().setMouthOpen(0);
   873→    useJarvisStore.getState().setAgentState('idle');
   874→  };
   875→
   876→  let idx = 0;
   877→  const speakNext = () => {
   878→    if (idx >= chunks.length) { cleanup(); return; }
   879→    const utt = new SpeechSynthesisUtterance(chunks[idx]);
   880→    // Select a good English voice
   881→    const voices = synth.getVoices();
   882→    const preferred = voices.find(v => v.lang.startsWith('en') && v.name.includes('Google')) ||
   883→                      voices.find(v => v.lang.startsWith('en') && !v.localService) ||
   884→                      voices.find(v => v.lang.startsWith('en'));
   885→    if (preferred) utt.voice = preferred;
   886→
   887→    utt.onboundary = () => {
   888→      useJarvisStore.getState().setMouthOpen(0.5);
   889→      if (mouthDecay) clearTimeout(mouthDecay);
   890→      mouthDecay = setTimeout(() => { useJarvisStore.getState().setMouthOpen(0); }, 200);
   891→    };
   892→    utt.onend = () => { idx++; speakNext(); };
   893→    utt.onerror = () => { cleanup(); };
   894→    synth.speak(utt);
   895→  };
   896→  speakNext();
   897→}

   899→/* ═══════════════════════════════════════════════════════════════
   900→   MAIN PAGE
   901→   ═══════════════════════════════════════════════════════════════ */
   902→
   903→export default function Home() {
   904→  const [wakeWordActive, setWakeWordActive] = useState(false);
   905→  const [installPrompt, setInstallPrompt] = useState<Event | null>(null);
   906→  const [securityLevel, setSecurityLevelState] = useState<SecurityLevel>('none');
   907→  const wakeWordRef = useRef<WakeWordDetector | null>(null);
   908→  const analyserRef = useRef<AnalyserNode | null>(null);
   909→  const bargeInRef = useRef<(() => void) | null>(null);
   910→  const backchannelTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   911→
   912→  // PWA install handler
   913→  const handleInstallClick = useCallback(() => {
   914→    if (installPrompt) {
   915→      (installPrompt as any).prompt();
   916→      setInstallPrompt(null);
   917→    }
   918→  }, [installPrompt]);
   919→
   920→  // Security level cycle: none -> biometric -> voice -> both -> none
   921→  const cycleSecurity = useCallback(() => {
   922→    const levels: SecurityLevel[] = ['none', 'biometric', 'voice', 'both'];
   923→    const current = getSecurityLevel();
   924→    const next = levels[(levels.indexOf(current) + 1) % levels.length];
   925→    setSecurityLevel(next);
   926→    setSecurityLevelState(next);
   927→  }, []);
   928→
   929→  // Initialize connection + audio + PWA
   930→  useEffect(() => {
   931→    useJarvisStore.getState().setConnected(true);
   932→
   933→    // Create audio analyser for visualization
   934→    createAudioAnalyser().then(analyser => {
   935→      analyserRef.current = analyser;
   936→    }).catch(() => {});
   937→
   938→    // Register service worker for PWA
   939→    if ('serviceWorker' in navigator) {
   940→      navigator.serviceWorker.register('/sw.js').catch(() => {});
   941→    }
   942→
   943→    // PWA install prompt
   944→    const handleBeforeInstall = (e: Event) => { e.preventDefault(); setInstallPrompt(e); };
   945→    window.addEventListener('beforeinstallprompt', handleBeforeInstall);
   946→
   947→    // Load voices (needed for TTS selection)
   948→    speechSynthesis.getVoices();
   949→    speechSynthesis.onvoiceschanged = () => speechSynthesis.getVoices();
   950→
   951→    // Keyboard shortcut: Space to toggle mic (when not in textarea)
   952→    const handleKeyDown = (e: KeyboardEvent) => {
   953→      const tag = (e.target as HTMLElement).tagName;
   954→      if (tag === 'TEXTAREA' || tag === 'INPUT') return;
   955→      if (e.code === 'Space') {
   956→        e.preventDefault();
   957→        // Trigger mic toggle via clicking the mic button
   958→        document.querySelector<HTMLButtonElement>('[data-mic-btn]')?.click();
   959→      }
   960→    };
   961→    window.addEventListener('keydown', handleKeyDown);
   962→
   963→    return () => {
   964→      window.removeEventListener('keydown', handleKeyDown);
   965→      window.removeEventListener('beforeinstallprompt', handleBeforeInstall);
   966→      window.speechSynthesis.cancel();
   967→      wakeWordRef.current?.destroy();
   968→      bargeInRef.current?.();
   969→      cleanupAudioSystem();
   970→    };
   971→  }, []);
   972→
   973→  // Barge-in: stop speaking when user starts talking
   974→  useEffect(() => {
   975→    const { agentState } = useJarvisStore.getState();
   976→    if (agentState === 'speaking' && analyserRef.current) {
   977→      bargeInRef.current = startBargeInDetection(analyserRef.current, () => {
   978→        stopSpeaking();
   979→        playSoundEffect('click');
   980→      });
   981→    }
   982→    return () => { bargeInRef.current?.(); bargeInRef.current = null; };
   983→  });
   984→
   985→  // Backchanneling: brief affirmations while user is speaking
   986→  useEffect(() => {
   987→    const BACKCHANNEL_PHRASES = ['I see', 'right', 'yes', 'go on', 'mm-hmm', 'understood', 'continue'];
   988→    let active = false;
   989→
   990→    function scheduleBackchannel() {
   991→      if (backchannelTimerRef.current) clearTimeout(backchannelTimerRef.current);
   992→      const delay = 4000 + Math.random() * 3000; // 4–7 seconds
   993→      backchannelTimerRef.current = setTimeout(() => {
   994→        if (!active) return;
   995→        const state = useJarvisStore.getState().agentState;
   996→        if (state !== 'listening') { active = false; return; }
   997→
   998→        const phrase = BACKCHANNEL_PHRASES[Math.floor(Math.random() * BACKCHANNEL_PHRASES.length)];
   999→
  1000→        // Play subtle click sound before backchannel
  1001→        playSoundEffect('click');
  1002→
  1003→        // Speak backchannel at low volume, quick voice — without changing agentState
  1004→        const utt = new SpeechSynthesisUtterance(phrase);
  1005→        utt.volume = 0.3;
  1006→        utt.rate = 1.3;
  1007→        utt.pitch = 1.1;
  1008→        const voices = speechSynthesis.getVoices();
  1009→        const preferred = voices.find(v => v.lang.startsWith('en') && v.name.includes('Google')) ||
  1010→                          voices.find(v => v.lang.startsWith('en') && !v.localService) ||
  1011→                          voices.find(v => v.lang.startsWith('en'));
  1012→        if (preferred) utt.voice = preferred;
  1013→        speechSynthesis.speak(utt);
  1014→
  1015→        // Schedule next backchannel
  1016→        if (active) scheduleBackchannel();
  1017→      }, delay);
  1018→    }
  1019→
  1020→    function stopBackchannel() {
  1021→      active = false;
  1022→      if (backchannelTimerRef.current) {
  1023→        clearTimeout(backchannelTimerRef.current);
  1024→        backchannelTimerRef.current = null;
  1025→      }
  1026→    }
  1027→
  1028→    // Subscribe to agentState changes
  1029→    const unsub = useJarvisStore.subscribe((state, prev) => {
  1030→      if (state.agentState === 'listening' && prev.agentState !== 'listening') {
  1031→        active = true;
  1032→        scheduleBackchannel();
  1033→      } else if (state.agentState !== 'listening' && prev.agentState === 'listening') {
  1034→        stopBackchannel();
  1035→      }
  1036→    });
  1037→
  1038→    return () => {
  1039→      stopBackchannel();
  1040→      unsub();
  1041→    };
  1042→  }, []);
  1043→
  1044→  // Wake word toggle
  1045→  const toggleWakeWord = useCallback(() => {
  1046→    if (wakeWordActive) {
  1047→      wakeWordRef.current?.destroy();
  1048→      wakeWordRef.current = null;
  1049→      setWakeWordActive(false);
  1050→    } else {
  1051→      const detector = new WakeWordDetector();
  1052→      detector.start(
  1053→        () => {
  1054→          // Wake word detected!
  1055→          playSoundEffect('wake');
  1056→          stopSpeaking(); // Stop any ongoing TTS
  1057→          const store = useJarvisStore.getState();
  1058→          if (store.agentState !== 'thinking') {
  1059→            store.setAgentState('listening');
  1060→            // Auto-start STT for 15 seconds of continuous listening
  1061→            const SR = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition;
  1062→            if (!SR) return;
  1063→            const rec = new SR();
  1064→            rec.continuous = true; rec.interimResults = true; rec.lang = 'en-US';
  1065→            let timeout = setTimeout(() => {
  1066→              try { rec.stop(); } catch { /* */ }
  1067→              store.setAgentState('idle'); store.setTranscript('');
  1068→            }, 15000);
  1069→            rec.onresult = (ev: any) => {
  1070→              let interim = '', final = '';
  1071→              for (let i = ev.resultIndex; i < ev.results.length; i++) {
  1072→                const t = ev.results[i][0].transcript;
  1073→                if (ev.results[i].isFinal) final += t; else interim += t;
  1074→              }
  1075→              store.setTranscript(interim);
  1076→              if (final.trim()) {
  1077→                clearTimeout(timeout);
  1078→                store.setTranscript('');
  1079→                try { rec.stop(); } catch { /* */ }
  1080→                sendToJarvis(final.trim());
  1081→              }
  1082→            };
  1083→            rec.onerror = () => { clearTimeout(timeout); store.setAgentState('idle'); store.setTranscript(''); };
  1084→            rec.onend = () => { clearTimeout(timeout); if (store.agentState === 'listening') { store.setAgentState('idle'); store.setTranscript(''); } };
  1085→            try { rec.start(); } catch { /* */ }
  1086→          }
  1087→        },
  1088→        (text) => { /* interim transcript available for display */ },
  1089→        () => { /* error */ },
  1090→      );
  1091→      wakeWordRef.current = detector;
  1092→      setWakeWordActive(true);
  1093→    }
  1094→  }, [wakeWordActive]);
  1095→
  1096→  return (
  1097→    <div className="flex h-screen flex-col overflow-hidden bg-[#050810] text-white">
  1098→      <TooltipProvider delayDuration={300}>
  1099→        <StatusBar wakeWordActive={wakeWordActive} onToggleWakeWord={toggleWakeWord} installPrompt={installPrompt} onInstallClick={handleInstallClick} securityLevel={securityLevel} onCycleSecurity={cycleSecurity} />
  1100→        <main className="flex flex-1 flex-col overflow-hidden md:flex-row">
  1101→          <FacePanel analyser={analyserRef.current} />
  1102→          <div className="hidden md:block w-px bg-gradient-to-b from-transparent via-cyan-500/10 to-transparent" />
  1103→          <ChatPanel />
  1104→        </main>
  1105→      </TooltipProvider>
  1106→    </div>
  1107→  );
  1108→}
