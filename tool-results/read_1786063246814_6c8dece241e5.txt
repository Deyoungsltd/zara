   506→function sendToJarvis(text: string, image?: string) {
   507→  const store = useJarvisStore.getState();
   508→  playSoundEffect('send');
   509→  store.addMessage({ id: crypto.randomUUID(), role: 'user', text, image: image || undefined, timestamp: Date.now() });
   510→  store.setAgentState('thinking');
   511→
   512→  const apiMessages = store.messages
   513→    .filter(m => m.role === 'user' || m.role === 'assistant')
   514→    .slice(-10)
   515→    .map(m => ({ role: m.role, content: m.text }));
   516→
   517→  // Inject memory context as system-like messages at the front
   518→  const memCtx = getMemoryContext();
   519→  if (memCtx) {
   520→    apiMessages.unshift({ role: 'user', content: `[Memory context: ${memCtx}]` });
   521→    apiMessages.unshift({ role: 'assistant', content: 'Understood, I will use this context.' });
   522→  }
   523→
   524→  const body: Record<string, any> = { messages: apiMessages };
   525→  if (image) body.image = image;
   526→
   527→  fetch('/api/jarvis', {
   528→    method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
   529→  })
   530→    .then(r => r.json())
   531→    .then(data => {
   532→      if (data.error) {
   533→        store.addMessage({ id: crypto.randomUUID(), role: 'system', text: `Error: ${data.error}`, timestamp: Date.now() });
   534→        store.setAgentState('idle');
   535→        playSoundEffect('error');
   536→        return;
   537→      }
   538→      if (data.actionRequest) {
   539→        store.setPendingAction({
   540→          actionId: crypto.randomUUID(), tier: data.actionRequest.tier || 'T2',
   541→          tool: data.actionRequest.tool || 'unknown', args: data.actionRequest.args || {},
   542→          description: data.actionRequest.description || 'Action requested',
   543→        });
   544→        store.setAgentState('awaiting_confirmation');
   545→        return;
   546→      }
   547→      if (data.text) {
   548→        store.addMessage({ id: crypto.randomUUID(), role: 'assistant', text: data.text, timestamp: Date.now() });
   549→        playSoundEffect('receive');
   550→        speakTextWithMouthSync(data.text);
   551→      }
   552→      if (data.toolExecutions) {
   553→        data.toolExecutions.forEach((te: any) => store.setLastToolExec({ tool: te.tool, result: te.result, tier: te.tier }));
   554→      }
   555→      // Persist conversation
   556→      const msgs = store.messages.filter(m => m.role === 'user' || m.role === 'assistant');
   557→      saveConversation(msgs.map(m => ({ role: m.role, text: m.text, timestamp: m.timestamp })));
   558→    })
   559→    .catch(() => {
   560→      store.addMessage({ id: crypto.randomUUID(), role: 'system', text: 'Connection error. Check your connection.', timestamp: Date.now() });
   561→      store.setAgentState('idle');
   562→      playSoundEffect('error');
   563→    });
   564→}

   849→
   850→function speakTextWithMouthSync(text: string) {
   851→  const synth = window.speechSynthesis;
   852→  if (!synth) return;
   853→  synth.cancel();
   854→
   855→  const store = useJarvisStore.getState();
   856→  store.setAgentState('speaking');
   857→
   858→  const chunks: string[] = [];
   859→  const sentences = text.match(/[^.!?\n]+[.!?\n]+/g) || [text];
   860→  let buffer = '';
   861→  for (const s of sentences) {
   862→    if ((buffer + s).length > 180 && buffer.trim()) { chunks.push(buffer.trim()); buffer = s; } else { buffer += s; }
   863→  }
   864→  if (buffer.trim()) chunks.push(buffer.trim());
   865→
   866→  let finished = false;
   867→  let mouthDecay: ReturnType<typeof setTimeout> | null = null;
   868→
   869→  const cleanup = () => {
   870→    if (finished) return;
   871→    finished = true;
   872→    if (mouthDecay) clearTimeout(mouthDecay);
   873→    try { synth.cancel(); } catch { /* */ }
   874→    useJarvisStore.getState().setMouthOpen(0);
   875→    useJarvisStore.getState().setAgentState('idle');
   876→  };
   877→
   878→  let idx = 0;
   879→  const speakNext = () => {
   880→    if (finished || idx >= chunks.length) { cleanup(); return; }
   881→    const utt = new SpeechSynthesisUtterance(chunks[idx]);
   882→
   883→    // Pick best available voice
   884→    const voices = synth.getVoices();
   885→    const preferred = voices.find(v => v.lang.startsWith('en') && v.name.includes('Google')) ||
   886→                      voices.find(v => v.lang.startsWith('en') && !v.localService) ||
   887→                      voices.find(v => v.lang.startsWith('en'));
   888→    if (preferred) utt.voice = preferred;
   889→    utt.rate = 1.0;
   890→    utt.pitch = 0.95;
   891→
   892→    utt.onboundary = () => {
   893→      useJarvisStore.getState().setMouthOpen(0.5);
   894→      if (mouthDecay) clearTimeout(mouthDecay);
   895→      mouthDecay = setTimeout(() => { useJarvisStore.getState().setMouthOpen(0); }, 180);
   896→    };
   897→    utt.onend = () => { idx++; speakNext(); };
   898→    utt.onerror = (e) => { console.warn('[TTS error]', e); cleanup(); };
   899→    synth.speak(utt);
   900→  };
   901→
   902→  // Mobile Chrome bug: speechSynthesis pauses after ~15s.
   903→  // Workaround: briefly pause+resume every 10s to keep it alive.
   904→  const keepalive = setInterval(() => {
   905→    if (finished) { clearInterval(keepalive); return; }
   906→    if (synth.speaking && !synth.paused) {
   907→      synth.pause();
   908→      setTimeout(() => { try { synth.resume(); } catch { /* */ } }, 25);
   909→    }
   910→  }, 10000);
   911→
   912→  // Safety timeout: force-stop after 60s total
   913→  const safetyTimeout = setTimeout(() => { cleanup(); clearInterval(keepalive); }, 60000);
   914→
   915→  speakNext();
   916→}

   986→      wakeWordRef.current?.destroy();
   987→      bargeInRef.current?.();
   988→      cleanupAudioSystem();
   989→    };
   990→  }, []);
   991→
   992→  // Barge-in: stop speaking when user starts talking
   993→  useEffect(() => {
   994→    const { agentState } = useJarvisStore.getState();
   995→    if (agentState === 'speaking' && analyserRef.current) {
   996→      bargeInRef.current = startBargeInDetection(analyserRef.current, () => {
   997→        stopSpeaking();
   998→        playSoundEffect('click');
   999→      });
  1000→    }
  1001→    return () => { bargeInRef.current?.(); bargeInRef.current = null; };
  1002→  });
  1003→
  1004→  // Backchanneling: brief affirmations while user is speaking
  1005→  // Disabled on mobile — causes TTS conflicts
  1006→  useEffect(() => {
  1007→    const isMobile = /iPhone|iPad|iPod|Android/i.test(navigator.userAgent);
  1008→    if (isMobile) return;
  1009→
  1010→    const BACKCHANNEL_PHRASES = ['I see', 'right', 'yes', 'go on', 'mm-hmm', 'understood', 'continue'];
  1011→    let active = false;
  1012→
  1013→    function scheduleBackchannel() {
  1014→      if (backchannelTimerRef.current) clearTimeout(backchannelTimerRef.current);
  1015→      const delay = 4000 + Math.random() * 3000; // 4–7 seconds
  1016→      backchannelTimerRef.current = setTimeout(() => {
  1017→        if (!active) return;
  1018→        const state = useJarvisStore.getState().agentState;
  1019→        if (state !== 'listening') { active = false; return; }
  1020→
  1021→        const phrase = BACKCHANNEL_PHRASES[Math.floor(Math.random() * BACKCHANNEL_PHRASES.length)];
  1022→
  1023→        // Play subtle click sound before backchannel
  1024→        playSoundEffect('click');
  1025→
  1026→        // Speak backchannel at low volume, quick voice — without changing agentState
  1027→        const utt = new SpeechSynthesisUtterance(phrase);
  1028→        utt.volume = 0.3;
  1029→        utt.rate = 1.3;
  1030→        utt.pitch = 1.1;
  1031→        const voices = speechSynthesis.getVoices();
  1032→        const preferred = voices.find(v => v.lang.startsWith('en') && v.name.includes('Google')) ||
  1033→                          voices.find(v => v.lang.startsWith('en') && !v.localService) ||
  1034→                          voices.find(v => v.lang.startsWith('en'));
  1035→        if (preferred) utt.voice = preferred;
  1036→        speechSynthesis.speak(utt);
  1037→
  1038→        // Schedule next backchannel
  1039→        if (active) scheduleBackchannel();
  1040→      }, delay);
  1041→    }
  1042→
  1043→    function stopBackchannel() {
  1044→      active = false;
  1045→      if (backchannelTimerRef.current) {
  1046→        clearTimeout(backchannelTimerRef.current);
  1047→        backchannelTimerRef.current = null;
  1048→      }
  1049→    }
  1050→
  1051→    // Subscribe to agentState changes
  1052→    const unsub = useJarvisStore.subscribe((state, prev) => {
  1053→      if (state.agentState === 'listening' && prev.agentState !== 'listening') {
  1054→        active = true;
  1055→        scheduleBackchannel();
  1056→      } else if (state.agentState !== 'listening' && prev.agentState === 'listening') {
  1057→        stopBackchannel();
  1058→      }
  1059→    });
  1060→
  1061→    return () => {
  1062→      stopBackchannel();
  1063→      unsub();
  1064→    };
  1065→  }, []);
  1066→
  1067→  // Wake word toggle
  1068→  const toggleWakeWord = useCallback(() => {
  1069→    if (wakeWordActive) {
  1070→      wakeWordRef.current?.destroy();
  1071→      wakeWordRef.current = null;
  1072→      setWakeWordActive(false);
  1073→    } else {
  1074→      const detector = new WakeWordDetector();
  1075→      detector.start(
  1076→        () => {
  1077→          // Wake word detected!
  1078→          playSoundEffect('wake');
  1079→          stopSpeaking(); // Stop any ongoing TTS
  1080→          const store = useJarvisStore.getState();
  1081→          if (store.agentState !== 'thinking') {
  1082→            store.setAgentState('listening');
  1083→            // Auto-start STT for 15 seconds of continuous listening
  1084→            const SR = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition;
  1085→            if (!SR) return;
  1086→            const rec = new SR();
  1087→            rec.continuous = true; rec.interimResults = true; rec.lang = 'en-US';
  1088→            let timeout = setTimeout(() => {
  1089→              try { rec.stop(); } catch { /* */ }
  1090→              store.setAgentState('idle'); store.setTranscript('');
  1091→            }, 15000);
  1092→            rec.onresult = (ev: any) => {
  1093→              let interim = '', final = '';
  1094→              for (let i = ev.resultIndex; i < ev.results.length; i++) {
  1095→                const t = ev.results[i][0].transcript;
  1096→                if (ev.results[i].isFinal) final += t; else interim += t;
  1097→              }
  1098→              store.setTranscript(interim);
  1099→              if (final.trim()) {
  1100→                clearTimeout(timeout);
  1101→                store.setTranscript('');
  1102→                try { rec.stop(); } catch { /* */ }
  1103→                sendToJarvis(final.trim());
  1104→              }
  1105→            };
  1106→            rec.onerror = () => { clearTimeout(timeout); store.setAgentState('idle'); store.setTranscript(''); };
  1107→            rec.onend = () => { clearTimeout(timeout); if (store.agentState === 'listening') { store.setAgentState('idle'); store.setTranscript(''); } };
  1108→            try { rec.start(); } catch { /* */ }
  1109→          }
  1110→        },
  1111→        (text) => { /* interim transcript available for display */ },
  1112→        () => { /* error */ },
  1113→      );
  1114→      wakeWordRef.current = detector;
  1115→      setWakeWordActive(true);
  1116→    }
  1117→  }, [wakeWordActive]);
  1118→
  1119→  return (
  1120→    <div className="flex h-screen flex-col overflow-hidden bg-[#050810] text-white">
  1121→      <TooltipProvider delayDuration={300}>
  1122→        <StatusBar wakeWordActive={wakeWordActive} onToggleWakeWord={toggleWakeWord} installPrompt={installPrompt} onInstallClick={handleInstallClick} securityLevel={securityLevel} onCycleSecurity={cycleSecurity} />
  1123→        <main className="flex flex-1 flex-col overflow-hidden md:flex-row">
  1124→          <FacePanel analyser={analyserRef.current} />
  1125→          <div className="hidden md:block w-px bg-gradient-to-b from-transparent via-cyan-500/10 to-transparent" />
  1126→          <ChatPanel />
  1127→        </main>
  1128→      </TooltipProvider>
  1129→    </div>
  1130→  );
  1131→}
